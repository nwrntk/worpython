print และตัวแปร

สั่งให้โปรแกรมพูด และให้มันจำค่าไว้ใช้ต่อ — สองอย่างแรกที่ต้องทำเป็น

1.1 print() — แสดงผลออกหน้าจอ

print("สวัสดีชาวโลก")
# สวัสดีชาวโลก

ข้อความต้องอยู่ในเครื่องหมายคำพูด จะใช้ " หรือ ' ก็ได้ แต่ต้องเปิด-ปิดให้เป็นคู่เดียวกัน ส่วนตัวเลขไม่ต้องใส่

print("Python")     # Python
print('Python')     # Python  ← เหมือนกัน
print(2026)         # 2026    ← ตัวเลข ไม่ต้องมีคำพูด
print(2 + 3)        # 5       ← คำนวณให้ก่อนแล้วค่อยพิมพ์
print("2 + 3")      # 2 + 3   ← มีคำพูด = ข้อความธรรมดา
print()             #         ← บรรทัดว่าง

1.2 พิมพ์หลายค่าในบรรทัดเดียว

คั่นด้วยจุลภาค Python จะใส่ช่องว่างให้อัตโนมัติ

print("อายุ", 20, "ปี")
# อายุ 20 ปี

เปลี่ยนตัวคั่นด้วย sep และเปลี่ยนตัวปิดท้ายด้วย end

print("2026", "08", "07", sep="-")
# 2026-08-07

print("กำลังโหลด", end="")
print("...เสร็จ")
# กำลังโหลด...เสร็จ   ← อยู่บรรทัดเดียวกัน
ปกติ print จะขึ้นบรรทัดใหม่ให้เสมอ เพราะค่าเริ่มต้นคือ end="\n" ถ้าไม่อยากให้ขึ้นบรรทัดใหม่ ให้ใส่ end=""

1.3 ตัวแปร — ให้โปรแกรมจำค่า

ตัวแปรคือชื่อที่เราตั้งให้กับค่าหนึ่ง เพื่อเอาไปใช้ซ้ำได้

name = "ฟ้าใส"
age = 20
gpa = 3.45

print(name)              # ฟ้าใส
print(name, age, gpa)    # ฟ้าใส 20 3.45

เครื่องหมาย = คือ "เอาค่าทางขวา ใส่ให้ชื่อทางซ้าย" ไม่ใช่ "เท่ากับ" แบบคณิตศาสตร์

x = 5
x = x + 3     # เอาค่า x เดิม (5) บวก 3 แล้วใส่กลับเข้า x
print(x)      # 8

ค่าใหม่ทับค่าเก่าได้เสมอ และเปลี่ยนชนิดข้อมูลก็ได้ด้วย

score = 80
score = 95        # ค่าเดิมหายไป เหลือ 95
score = "ดีมาก"    # เปลี่ยนจากตัวเลขเป็นข้อความก็ยังได้

1.5 input() — รับค่าจากผู้ใช้

print("สวัสดี", name)

# คุณชื่ออะไร: ฟ้าใส
# สวัสดี ฟ้าใส
สิ่งที่ input() คืนมาเป็นข้อความเสมอ แม้ผู้ใช้จะพิมพ์ตัวเลข
age = input("อายุ: ") แล้ว age + 1 จะ error ต้องแปลงเป็นตัวเลขก่อนด้วย int() (บทที่ 2)

print("ปีหน้าอายุ", age + 1)

1.6 f-string — วิธีพิมพ์ที่อ่านง่ายที่สุด

ใส่ f หน้าเครื่องหมายคำพูด แล้วเอาชื่อตัวแปรใส่ในปีกกาได้เลย

name = "ฟ้าใส"
age = 20

print(f"{name} อายุ {age} ปี")
# ฟ้าใส อายุ 20 ปี

# คำนวณในปีกกาได้ด้วย
print(f"ปีหน้าอายุ {age + 1} ปี")

เทียบกับวิธีอื่น — f-string อ่านง่ายกว่าชัดเจน

print("ชื่อ " + name + " อายุ " + str(age))    # ต้องแปลง str เอง ยุ่งยาก
print("ชื่อ", name, "อายุ", age)                # คุมช่องว่างยาก
print(f"ชื่อ {name} อายุ {age}")                 # ← ใช้อันนี้

โจทย์ 1 — แนะนำตัว

รับชื่อ อายุ และสาขาวิชาจากผู้ใช้ แล้วพิมพ์ประโยคแนะนำตัวด้วย f-string พร้อมบอกว่าอีก 10 ปีจะอายุเท่าไร

▸ ดูเฉลย 
name = input("ชื่อ: ")
age = int(input("อายุ: "))
major = input("สาขา: ")

print(f"สวัสดีครับ ผม/ดิฉัน {name} อายุ {age} ปี เรียนสาขา{major}")
print(f"อีก 10 ปีจะอายุ {age + 10} ปี")

โจทย์ 2 — เครื่องคิดเลขง่าย ๆ

รับตัวเลข 2 ตัวจากผู้ใช้ แล้วพิมพ์ผลบวก ลบ คูณ หาร ในรูปแบบ 10 + 3 = 13

▸ ดูเฉลย 
a = float(input("ตัวที่ 1: "))
b = float(input("ตัวที่ 2: "))

print(f"{a} + {b} = {a + b}")
print(f"{a} - {b} = {a - b}")
print(f"{a} * {b} = {a * b}")
print(f"{a} / {b} = {a / b}")

โจทย์ 3 — คำนวณราคาสินค้า

รับชื่อสินค้า ราคาต่อชิ้น และจำนวนชิ้น แล้วพิมพ์ใบเสร็จที่บอก ราคารวม, VAT 7% และยอดสุทธิ

▸ ดูเฉลย 
item = input("ชื่อสินค้า: ")
price = float(input("ราคาต่อชิ้น: "))
qty = int(input("จำนวน: "))

subtotal = price * qty
vat = subtotal * 0.07
total = subtotal + vat

print("-" * 30)
print(f"{item} x {qty}")
print(f"ราคารวม   {subtotal:>10.2f}")
print(f"VAT 7%    {vat:>10.2f}")
print(f"ยอดสุทธิ  {total:>10.2f}")
print("-" * 30)

โจทย์ 4 — ท้าทาย: สลับค่าและวาดกรอบ

รับค่า 2 ค่าจากผู้ใช้ พิมพ์ก่อนสลับและหลังสลับ จากนั้นวาดกรอบล้อมข้อความด้วย * โดยกรอบต้องกว้างพอดีกับข้อความ (ใบ้: len() กับการคูณ string)

▸ ดูเฉลย 
a = input("ค่าที่ 1: ")
b = input("ค่าที่ 2: ")

print(f"ก่อนสลับ: a={a} b={b}")
a, b = b, a
print(f"หลังสลับ: a={a} b={b}")

msg = f"a={a} b={b}"
width = len(msg) + 4

print("*" * width)
print(f"* {msg} *")
print("*" * width)
