17.3 รูปภาพคือ array ของตัวเลข

img = cv2.imread("cat.jpg")

print(img.dtype)     # uint8 = จำนวนเต็ม 0 ถึง 255

h, w, c = img.shape

print(f"จุด (50,100) → B={b} G={g} R={r}")

OpenCV เก็บสีเป็น BGR ไม่ใช่ RGB — และ index เป็น [y, x] ไม่ใช่ [x, y]

ทำไมภาพถึงเป็นตัวเลข และ 0-255 มาจากไหน — เรื่อง sampling, quantization และรูปแบบภาพทั้ง 4 แบบ (เทา · ไบนารี · RGB · HSV) อยู่ใน บทที่ 18

17.6 เบลอ · Threshold · หาขอบ

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

blur = cv2.GaussianBlur(gray, (5, 5), 0)       # เบลอลดจุดรบกวน (ขนาดต้องเป็นเลขคี่)

# แยกขาว-ดำ: จุดที่สว่างกว่า 127 เป็นขาว(255) ที่เหลือเป็นดำ(0)
_, binary = cv2.threshold(blur, 127, 255, cv2.THRESH_BINARY)

edges = cv2.Canny(blur, 100, 200)              # หาเส้นขอบในภาพ

# นับจุดขาว — ภาพเป็น array เลยใช้ NumPy นับได้ตรง ๆ
white = int(np.sum(binary == 255))
print(f"จุดขาว {white} จุด จากทั้งหมด {binary.size} จุด")
ลำดับที่ใช้จริงเกือบทุกงาน — แปลงเป็นเทา → เบลอ → threshold หรือ Canny การเบลอก่อนสำคัญมาก เพราะจุดรบกวนเม็ดเล็ก ๆ จะกลายเป็นขอบปลอมเต็มไปหมด
เกณฑ์ 127 ไม่ต้องเดาเองก็ได้ — Otsu หา T ให้อัตโนมัติ, adaptive คิด T แยกทีละส่วนของภาพ, morphology เก็บกวาดเศษหลัง threshold และ convolution คือสิ่งที่อยู่เบื้องหลัง GaussianBlur กับ Canny — ทั้งหมดอยู่ใน บทที่ 18

17.7 กล้องและวิดีโอ

วิดีโอคือรูปหลาย ๆ ใบต่อกัน — อ่านทีละเฟรมด้วย while loop

cap = cv2.VideoCapture(0)          # 0 = กล้องตัวแรก, หรือใส่ "video.mp4"

if not cap.isOpened():
    print("เปิดกล้องไม่ได้")
    exit()

while True:
    ret, frame = cap.read()        # ret = True/False ว่าอ่านสำเร็จไหม
    if not ret:
        break                      # กล้องหลุด หรือวิดีโอจบ

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    cv2.imshow("camera", gray)

    if cv2.waitKey(1) & 0xFF == ord("q"):   # กด q เพื่อออก
        break

cap.release()                      # คืนกล้องให้ระบบ ห้ามลืม
cv2.destroyAllWindows()
ส่วนทำไมต้องมี
cap.isOpened()กล้องอาจถูกโปรแกรมอื่นยึดอยู่ เช็กก่อนวน
if not ret: breakวิดีโอจบแล้ว frame จะเป็น None
waitKey(1)1 มิลลิวินาที — ทั้งหน่วงเฟรมและรับปุ่มไปพร้อมกัน
& 0xFFตัดเอาเฉพาะ 8 บิตท้าย ให้เทียบกับ ord("q") ได้ตรง
cap.release()ไม่เรียก = ไฟกล้องค้างติด เปิดใหม่ไม่ได้จนกว่าจะปิดโปรแกรม
ลืม waitKey ใน loop = จอค้าง และ ลืม release = กล้องค้าง สองอย่างนี้คือบั๊กที่มือใหม่เจอบ่อยที่สุด

17.8 ตัวอย่างรวม — ตรวจจับใบหน้า

import cv2

# ไฟล์โมเดลติดมากับ OpenCV อยู่แล้ว ไม่ต้องโหลดเพิ่ม
path = cv2.data.haarcascades + "haarcascade_frontalface_default.xml"
detector = cv2.CascadeClassifier(path)

cap = cv2.VideoCapture(0)

while True:
    ret, frame = cap.read()
    if not ret:
        break

    gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    faces = detector.detectMultiScale(gray, 1.1, 5)   # คืน list ของ (x, y, w, h)

    for (x, y, w, h) in faces:                        # วน list แบบเดียวกับบทที่ 6
        cv2.rectangle(frame, (x, y), (x + w, y + h), (0, 255, 0), 2)

    cv2.putText(frame, f"faces: {len(faces)}", (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
    cv2.imshow("face detect", frame)

    if cv2.waitKey(1) & 0xFF == ord("q"):
        break

cap.release()
cv2.destroyAllWindows()
ตัวเลข 2 ตัวใน detectMultiScale คือปุ่มปรับ — 1.1 คือความละเอียดในการค้นหา (ยิ่งใกล้ 1 ยิ่งแม่นแต่ช้า) และ 5 คือความเข้มงวด (ยิ่งมากยิ่งเจอน้อยลงแต่มั่นใจกว่า) ต้องลองปรับตามแสงและกล้องจริง ไม่มีค่าที่ดีที่สุดค่าเดียว

โจทย์ 1 — เปิดรูปแบบไม่พัง

เขียนฟังก์ชัน open_image(path) ที่อ่านรูปแล้วคืน array ถ้าเปิดไม่ได้ให้พิมพ์บอกและคืน None จากนั้นใช้ฟังก์ชันนี้เปิดรูป แล้วพิมพ์ความกว้าง ความสูง และจำนวนช่องสี

▸ ดูเฉลย 
import cv2

def open_image(path):
    img = cv2.imread(path)
    if img is None:
        print(f"เปิดไม่ได้: {path}")
        return None
    return img

img = open_image("cat.jpg")

if img is not None:
    h, w, c = img.shape
    print(f"กว้าง {w} สูง {h} ช่องสี {c}")
    cv2.imshow("image", img)
    cv2.waitKey(0)
    cv2.destroyAllWindows()

โจทย์ 2 — โรงงานแปลงรูป

รับรูปมา 1 ใบ สร้างออกมา 4 เวอร์ชันแล้วบันทึกลงไฟล์: ขาวดำ, ย่อครึ่ง, กลับซ้ายขวา และภาพขอบ (Canny) โดยใช้ list ของคู่ (ชื่อไฟล์, รูป) แล้ว for วนบันทึกทีเดียว

▸ ดูเฉลย 
import cv2

img = cv2.imread("cat.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

outputs = [
    ("out_gray.jpg",  gray),
    ("out_small.jpg", cv2.resize(img, None, fx=0.5, fy=0.5)),

for name, image in outputs:

โจทย์ 3 — วัดความสว่างของรูป

อ่านรูป แปลงเป็นขาวดำ หาค่าเฉลี่ยความสว่าง (0-255) แล้วบอกว่ารูป มืด (< 85), ปกติ (85-170) หรือ สว่าง (> 170) พร้อมเขียนผลลัพธ์ทับลงบนรูปด้วย putText

▸ ดูเฉลย 
import cv2

img = cv2.imread("cat.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
mean = gray.mean()            # ค่าเฉลี่ยของทุกจุดในภาพ

if mean < 85:
    label = "DARK"
elif mean <= 170:
    label = "NORMAL"
else:
    label = "BRIGHT"

print(f"ความสว่างเฉลี่ย {mean:.1f} → {label}")

cv2.putText(img, f"{label} ({mean:.0f})", (10, 30),
            cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
cv2.imshow("brightness", img)
cv2.waitKey(0)
cv2.destroyAllWindows()

โจทย์ 4 — ท้าทาย: กล้องถ่ายภาพพร้อมฟิลเตอร์

ทำโปรแกรมกล้องที่กดปุ่มเปลี่ยนโหมดได้ — รวม while, if/elif และการเขียนไฟล์:

  1. กด 1 ภาพปกติ · 2 ขาวดำ · 3 ภาพขอบ
  2. กด s บันทึกเฟรมปัจจุบันเป็น shot1.jpg, shot2.jpg, ... ไม่เขียนทับกัน
  3. กด q ออก และแสดงชื่อโหมดปัจจุบันบนภาพเสมอ
▸ ดูเฉลย 
import cv2

cap = cv2.VideoCapture(0)
if not cap.isOpened():
    print("เปิดกล้องไม่ได้")
    exit()

mode = "normal"
count = 0

while True:
    ret, frame = cap.read()
    if not ret:
        break

    if mode == "gray":
        show = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
    elif mode == "edge":
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        show = cv2.Canny(cv2.GaussianBlur(gray, (5, 5), 0), 100, 200)
    else:
        show = frame.copy()

    cv2.putText(show, mode, (10, 30),
                cv2.FONT_HERSHEY_SIMPLEX, 0.8, (0, 255, 0), 2)
    cv2.imshow("camera", show)

    key = cv2.waitKey(1) & 0xFF
    if key == ord("q"):
        break
    elif key == ord("1"):
        mode = "normal"
    elif key == ord("2"):
        mode = "gray"
    elif key == ord("3"):
        mode = "edge"
    elif key == ord("s"):
        count += 1
        name = f"shot{count}.jpg"
        cv2.imwrite(name, show)
        print("บันทึก", name)

cap.release()
cv2.destroyAllWindows()
print(f"ถ่ายไปทั้งหมด {count} รูป")
OpenCV คือประตูบานแรกของงาน Computer Vision — ต่อจากนี้มี mediapipe (จับมือและท่าทาง), pytesseract (อ่านตัวอักษรจากภาพ) และ ultralytics (YOLO ตรวจจับวัตถุ) ซึ่งทั้งหมดรับ-ส่งภาพเป็น array หน้าตาเดียวกับที่เรียนในบทนี้ — แต่ก่อนไปถึงตรงนั้น บทที่ 18 จะอธิบายว่าคำสั่งที่เพิ่งใช้ไปทำอะไรกับตัวเลขในภาพบ้าง
