12.2 return — ส่งค่ากลับ

ความต่างที่คนใหม่สับสนบ่อยที่สุด: print คือแสดง ส่วน return คือส่งค่ากลับไปให้คนเรียกใช้ต่อ

def add_print(a, b):
    print(a + b)        # แค่แสดงผล

def add_return(a, b):
    return a + b        # ส่งค่ากลับ

x = add_print(2, 3)     # พิมพ์ 5 ออกมา แต่ x = None
y = add_return(2, 3)    # ไม่พิมพ์อะไร แต่ y = 5

print(y * 10)           # 50 ← เอาไปใช้ต่อได้
ฟังก์ชันที่ไม่มี return จะคืน None เสมอ
และเมื่อเจอ return ฟังก์ชันจะจบทันที โค้ดบรรทัดหลังจากนั้นไม่ทำงาน 
def check_pass(score):
    if score >= 50:
        return "ผ่าน"       # จบตรงนี้เลย
    return "ไม่ผ่าน"        # ทำงานเฉพาะตอนไม่เข้า if

print(check_pass(80))       # ผ่าน

12.5 ขอบเขตตัวแปร (Scope)

ตัวแปรที่สร้างในฟังก์ชันเป็น local — อยู่แค่ในฟังก์ชันนั้น พอฟังก์ชันจบก็หายไป

    x = 10          # local — มีอยู่แค่ในนี้

12.8 ประกอบร่างทุกบท

def grade_report(scores):
    """รับ dict ชื่อ→คะแนน คืน dict ชื่อ→เกรด"""
    result = {}
    for name, s in scores.items():
        if   s >= 80: g = "A"
        elif s >= 70: g = "B"
        elif s >= 60: g = "C"
        elif s >= 50: g = "D"
        else:         g = "F"
        result[name] = g
    return result

scores = {"ฟ้า": 85, "นล": 72, "อาร์ม": 45}

for name, g in grade_report(scores).items():
    print(f"{name:8s} {scores[name]:3d} → {g}")

12.10 import — ใช้ฟังก์ชันที่มีคนเขียนไว้แล้ว

ฟังก์ชันที่เราเขียนเก็บไว้ในไฟล์ได้ ไฟล์ .py ที่รวมฟังก์ชันไว้ให้คนอื่นเรียกใช้เรียกว่า module · Python มี module มาให้หลายร้อยตัว (เรียกว่า standard library) แค่ import ก็ใช้ได้เลย ไม่ต้องเขียนเอง

โจทย์ 1 — ฟังก์ชันพื้นฐาน 3 ตัว

เขียนฟังก์ชัน 3 ตัวพร้อมทดสอบ:

  1. area_circle(r) — คืนพื้นที่วงกลม (ใช้ 3.14159)

def area_circle(r):
    return 3.14159 * r ** 2

def is_even(n):
    return n % 2 == 0        # เงื่อนไขคืน True/False อยู่แล้ว

def celsius_to_f(c):
    return c * 9 / 5 + 32

print(area_circle(3))        # 28.27431
print(is_even(7))            # False
print(celsius_to_f(100))     # 212.0

def add(a, b): return a + b
def sub(a, b): return a - b

def div(a, b):
    if b == 0:
        return "หารด้วยศูนย์ไม่ได้"
    return a / b

while True:
    op = input("\nเลือก + - * / (q เพื่อออก): ")

        break

    if   op == "+": print(add(a, b))

โจทย์ 3 — สถิติจาก list

เขียน stats(nums) ที่คืน (จำนวน, ผลรวม, ค่าเฉลี่ย, ค่ามากสุด, ค่าน้อยสุด) โดยห้ามใช้ sum(), max(), min() ให้เขียน loop เอง และถ้า list ว่างให้คืน None

▸ ดูเฉลย 
def stats(nums):
    if not nums:
        return None

    total = 0
    hi = nums[0]
    lo = nums[0]

    for n in nums:
        total += n
        if n > hi: hi = n
        if n < lo: lo = n

    return len(nums), total, total / len(nums), hi, lo

result = stats([78, 95, 62, 88])
if result is None:
    print("ไม่มีข้อมูล")
else:
    n, total, avg, hi, lo = result

โจทย์ 4 — ท้าทาย: ระบบเกรดแบบฟังก์ชัน

แยกโปรแกรมออกเป็นฟังก์ชันย่อย ๆ:

  • input_scores() — รับคะแนนจนกว่าจะใส่ค่าว่าง คืนเป็น dict ชื่อ→คะแนน
  • to_grade(score) — แปลงคะแนนเป็นเกรด A/B/C/D/F
  • summary(scores) — คืน (เฉลี่ย, ชื่อคนคะแนนสูงสุด, จำนวนคนที่สอบผ่าน)
  • show_report(scores) — พิมพ์ตารางรายงาน
▸ ดูเฉลย 
def input_scores():
    scores = {}
    while True:
        name = input("ชื่อ (เว้นว่างเพื่อจบ): ").strip()
        if name == "":
            break
        scores[name] = int(input(f"คะแนนของ {name}: "))
    return scores


def to_grade(score):
    if   score >= 80: return "A"
    elif score >= 70: return "B"
    elif score >= 60: return "C"
    elif score >= 50: return "D"
    return "F"


def summary(scores):
    avg = sum(scores.values()) / len(scores)
    top = max(scores, key=scores.get)
    passed = 0
    for s in scores.values():
        if s >= 50:
            passed += 1
    return avg, top, passed


def show_report(scores):
    if not scores:
        print("ไม่มีข้อมูล")
        return                      # return เปล่า = ออกจากฟังก์ชันเลย

    print(f"\n{'ชื่อ':<12}{'คะแนน':>8}{'เกรด':>6}")
    print("-" * 26)
    for name, s in scores.items():
        print(f"{name:<12}{s:>8}{to_grade(s):>6}")

    avg, top, passed = summary(scores)
    print("-" * 26)
    print(f"เฉลี่ย {avg:.2f} · สูงสุด {top} · ผ่าน {passed}/{len(scores)} คน")


show_report(input_scores())

โจทย์ 8 — ทอยลูกเต๋า 1,000 ครั้ง

ทอยลูกเต๋า 2 ลูก 1,000 ครั้ง นับว่าผลรวมแต่ละค่า (2 ถึง 12) ออกกี่ครั้ง แล้วพิมพ์เป็นกราฟดาว (ดาว 1 ดวงต่อ 10 ครั้ง) · ใช้ random กับการนับด้วย dict (บทที่ 9) แล้วสังเกตว่าผลรวมไหนออกบ่อยที่สุด

▸ ดูเฉลย 
import random

counts = {}
for _ in range(1000):
    total = random.randint(1, 6) + random.randint(1, 6)
    counts[total] = counts.get(total, 0) + 1

for s in sorted(counts):
    print(f"{s:2} {counts[s]:4} " + "*" * (counts[s] // 10))
