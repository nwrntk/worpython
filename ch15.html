15.1 ภาพดิจิทัลเกิดขึ้นได้ยังไง

ภาพจริงเป็นสัญญาณต่อเนื่อง กล้องแปลงเป็นตัวเลขด้วยสองขั้นตอน

ขั้นตอนทำอะไรถ้าน้อยเกินไป
Samplingแบ่งพื้นที่เป็นตารางพิกเซล N×Nภาพแตกเป็นบล็อกเหลี่ยม ๆ
Quantizationปัดค่าความสว่างให้เหลือ L ระดับไล่เฉดหายไป เกิดแถบสีเป็นชั้น ๆ

ผลลัพธ์คือเมทริกซ์ของจำนวนเต็ม — ไม่มีอะไรมากกว่านั้น ลองทำให้ดูทั้งสองขั้น

import cv2
import numpy as np

img = cv2.imread("cat.jpg")
h, w = img.shape[:2]

# sampling หยาบ: ย่อลง 10 เท่า แล้วขยายกลับแบบไม่เกลี่ยสี
small  = cv2.resize(img, None, fx=0.1, fy=0.1, interpolation=cv2.INTER_AREA)
blocky = cv2.resize(small, (w, h), interpolation=cv2.INTER_NEAREST)

# quantization: เหลือ 4 ระดับ จาก 256 ระดับ
L = 4
step = 256 // L
quant = (img // step) * step        # ปัดลงเป็นขั้น ๆ

cv2.imwrite("out_sampling.jpg", blocky)
cv2.imwrite("out_quant.jpg", quant)
ทั้งสองบรรทัดคือคณิตศาสตร์ล้วน ๆ — img // step * step คือการหารปัดลงแล้วคูณกลับ ที่เรียนมาตั้งแต่บทที่ 2 แค่ทำพร้อมกันทุกพิกเซลเพราะ NumPy คิดให้ทั้ง array ทีเดียว

15.2 รูปแบบภาพ 4 แบบ

แบบเก็บอะไรใช้เมื่อ
Grayscale1 ช่อง · 0-255งานส่วนใหญ่ — เร็วและพอแล้ว
Binary0 หรือ 255 เท่านั้นนับวัตถุ วัดพื้นที่ หาคอนทัวร์
RGB / BGR3 ช่อง · 16.7 ล้านสีภาพที่คนดู
HSVสี · ความสด · ความสว่างคัดแยกวัตถุตามสี
# เทา — ถ่วงน้ำหนักตามความไวของตาคน เขียวหนักสุด
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)   # 0.299R + 0.587G + 0.114B

# แยกดูทีละช่องสี — ท้องฟ้าจะสว่างในช่อง B ต้นหญ้าสว่างในช่อง G
b, g, r = cv2.split(img)

# ไบนารี — เทียบกับเกณฑ์ T
T = 127
_, binary = cv2.threshold(gray, T, 255, cv2.THRESH_BINARY)

# HSV — สร้างมาสก์เฉพาะสีเขียว
hsv  = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
mask = cv2.inRange(hsv, (35, 80, 40), (85, 255, 255))
only_green = cv2.bitwise_and(img, img, mask=mask)
Hue ของ OpenCV คือ 0-179 ไม่ใช่ 0-360 — เพราะต้องยัดลงใน 1 ไบต์ ค่าที่อ่านเจอในตำราต้องหารสองก่อนเสมอ (แดง 0 · เหลือง 30 · เขียว 60 · ฟ้า 90 · น้ำเงิน 120)
ทำไมต้อง HSV — เงาหรือแสงที่เปลี่ยนไปกระทบ V เป็นหลัก แต่ H แทบไม่ขยับ การคัดแยกวัตถุตามสีจึงทนแสงกว่าใช้ RGB มาก

hist = cv2.calcHist([gray], [0], None, [256], [0, 256])
print(hist.shape)                    # (256, 1)

# นับเองด้วย NumPy ก็ได้ ผลเท่ากัน
hist2 = np.bincount(gray.ravel(), minlength=256)

print("มืดสุด", gray.min(), "สว่างสุด", gray.max())
print("เฉลี่ย", round(gray.mean(), 1), "ส่วนเบี่ยงเบน", round(gray.std(), 1))
อยากเห็นฮิสโทแกรมเป็นกราฟจริง ๆ — วาดด้วย matplotlib ได้ในสามบรรทัด ดู บทที่ 16

คอนทราสต์คือความกว้างของช่วงค่าที่ภาพใช้จริง — ภาพหม่นคือค่ากองอยู่กลางกราฟ ยืดให้เต็ม 0-255 ได้ด้วย

stretched = cv2.normalize(gray, None, 0, 255, cv2.NORM_MINMAX)
ดันคอนทราสต์แรงเกินไปจะเกิด clipping — พิกเซลไปกองที่ 0 กับ 255 แล้วรายละเอียดตรงนั้นหายถาวร ยืดกลับไม่ได้อีก

15.4 การดำเนินการระดับพิกเซล

ผลลัพธ์ที่จุด (y, x) ขึ้นกับค่าที่จุด (y, x) ของภาพต้นทางเท่านั้น — บวก ลบ คูณ หาร ตรง ๆ

bright = cv2.add(img, 50)                    # สว่างขึ้น — เกิน 255 ตัดเหลือ 255
dark   = cv2.subtract(img, 50)               # มืดลง — ต่ำกว่า 0 ตัดเหลือ 0
diff   = cv2.absdiff(frame1, frame2)         # ต่างกันตรงไหน = จับการเคลื่อนไหว

อย่าใช้ img + 50 เฉย ๆ — ชนิดข้อมูลเป็น uint8 ค่า 250 + 50 จะวนกลับเป็น 44 จุดที่ควรสว่างสุดจะกลายเป็นดำ
cv2.add() ตัดที่ 255 ให้อัตโนมัติ (เรียกว่า saturate) ใช้ตัวนี้เสมอ

15.5 Contrast enhancement — ตารางแปลงค่า

วิธีที่เร็วที่สุดคือทำตาราง 256 ช่อง ไว้ล่วงหน้า แล้วยิงทุกพิกเซลผ่านตารางนั้น (เรียกว่า transfer function หรือ LUT) — เส้นยิ่งชันช่วงไหน คอนทราสต์ช่วงนั้นยิ่งถูกขยาย

# ยืดช่วง 100-180 ให้กลายเป็น 0-255 นอกช่วงถูกตัดทิ้ง
lo, hi = 100, 180
table = np.clip((np.arange(256) - lo) * 255 / (hi - lo), 0, 255).astype(np.uint8)
out = cv2.LUT(gray, table)

# gamma — ดึงรายละเอียดในเงาโดยไม่เผาส่วนสว่าง
gamma = 0.5
gtable = ((np.arange(256) / 255) ** gamma * 255).astype(np.uint8)
out2 = cv2.LUT(gray, gtable)

# ให้เครื่องเกลี่ยฮิสโทแกรมให้เอง
eq    = cv2.equalizeHist(gray)                                  # ทั้งภาพ
clahe = cv2.createCLAHE(clipLimit=2.0, tileGridSize=(8, 8)).apply(gray)  # แยกเป็นช่อง ๆ
LUT คือ dictionary ที่เรียนในบทที่ 9 ในรูปแบบเร็วที่สุด — คำนวณครั้งเดียว 256 ค่า แล้วเปิดตารางดูเอาสำหรับพิกเซลนับล้าน แทนที่จะคำนวณทีละจุด

15.6 Convolution — หัวใจของฟิลเตอร์ทุกตัว

เลื่อนเคอร์เนล (ตารางน้ำหนักเล็ก ๆ) ไปทีละพิกเซล คูณค่าที่ทับกันแล้วบวกรวม จากนั้นหารด้วยผลรวมน้ำหนักเพื่อคงความสว่างเดิม

ดูเป็นภาพ · เคอร์เนลเลื่อนทับภาพ

กรอบสีเข้มคือเคอร์เนล ตัวเลขเหลืองมุมช่องคือน้ำหนัก · ทุกตำแหน่ง: คูณค่าที่ทับกันทั้ง 9 คู่ บวกรวม หารด้วย 16 (ผลรวมน้ำหนัก) ได้ 1 ช่องในภาพออก · กด ⏵ เล่น เพื่อดูเลื่อนเอง · เปลี่ยนตัวเลขในเคอร์เนล = เปลี่ยนจากเบลอเป็นคมหรือเป็นหาขอบ

# เฉลี่ยเท่ากันหมด — เบลอแบบหยาบ ๆ
kernel = np.ones((3, 3), np.float32) / 9
mean = cv2.filter2D(img, -1, kernel)

# เกาส์เซียน — กลางหนัก ขอบเบา ลดจุดรบกวนโดยยังรักษารูปทรง
g = cv2.getGaussianKernel(5, 1.0)
print((g @ g.T * 273).round())      # ดูน้ำหนัก: ตรงกลางมากสุด แล้วลดตามระยะ

blur = cv2.GaussianBlur(img, (5, 5), 0)   # ใช้จริงเรียกตัวนี้ เร็วกว่า

# เปลี่ยนเคอร์เนล = เปลี่ยนฟิลเตอร์ เช่น ทำให้คมขึ้น
sharpen = np.array([[0, -1, 0],
                    [-1, 5, -1],
                    [0, -1, 0]], np.float32)
out = cv2.filter2D(img, -1, sharpen)
เบลอ · คม · หาขอบ · ตรวจลาย — ทั้งหมดคือ convolution ตัวเดียวกัน ต่างกันแค่ตัวเลขในเคอร์เนล นี่คือเหตุผลที่ CNN ในงาน AI ถึงชื่อ “Convolutional”

15.7 Binarization — หาเกณฑ์ T ให้อัตโนมัติ

# Otsu: ไล่ลอง T ทุกค่า 0-255 เลือกค่าที่แยกสองกลุ่มได้ขาดที่สุด
T, otsu = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
print("Otsu เลือก T =", T)          # เลข 0 ข้างหน้าถูกมองข้าม Otsu หาเอง

# Adaptive: คิด T ใหม่ในทุกหน้าต่างย่อย — ทนแสงตกไม่เท่ากัน
adap = cv2.adaptiveThreshold(gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
                             cv2.THRESH_BINARY, 31, 5)   # 31 = ขนาดหน้าต่าง (คี่)
เอกสารถ่ายรูปแล้วมีเงามือ ใช้ Otsu ไม่รอด — เพราะมันหา T ค่าเดียวสำหรับทั้งภาพ งานแบบนั้นต้อง adaptive เสมอ

15.8 Morphology — จัดการรูปทรงบนภาพไบนารี

คำสั่งกติกาใช้ทำอะไร
Erosionเป็น 1 เมื่อใต้ SE เป็น 1 ทุกช่องกัดขอบ วัตถุเล็กลง
Dilationเป็น 1 เมื่อมี 1 อย่างน้อยหนึ่งช่องพองวัตถุ เชื่อมรอยขาด
Openingกัดแล้วพองลบจุดรบกวนเล็ก ๆ นอกวัตถุ
Closingพองแล้วกัดอุดรูในตัววัตถุ
se = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))   # รูปทรงหัวแปรง

ero    = cv2.erode(binary, se)
dil    = cv2.dilate(binary, se)
opened = cv2.morphologyEx(binary, cv2.MORPH_OPEN,  se)
closed = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, se)
Opening แล้ว Closing คือคู่หูมาตรฐาน — ทำต่อจาก threshold เกือบทุกครั้ง เพื่อเก็บกวาดเศษก่อนเอาไปนับวัตถุ

15.9 Edge detection — ขอบคืออนุพันธ์

ขอบภาพคือบริเวณที่ความสว่างเปลี่ยนเร็ว Sobel ใช้เคอร์เนลสองตัวจับการเปลี่ยนแปลง แนวนอนและแนวตั้ง แล้วรวมเป็นขนาดเกรเดียนต์ G = |Gx| + |Gy|

gx = cv2.Sobel(gray, cv2.CV_64F, 1, 0, ksize=3)   # เปลี่ยนตามแนวนอน
gy = cv2.Sobel(gray, cv2.CV_64F, 0, 1, ksize=3)   # เปลี่ยนตามแนวตั้ง

mag   = cv2.convertScaleAbs(np.abs(gx) + np.abs(gy))    # ความแรงของขอบ
angle = np.arctan2(gy, gx) * 180 / np.pi                # ทิศทางของขอบ (องศา)

lap = cv2.Laplacian(gray, cv2.CV_64F)             # อนุพันธ์อันดับสอง
ต้องใส่ cv2.CV_64F ไม่ใช่ปล่อยเป็น uint8 — เกรเดียนต์มีค่าติดลบ (ขอบขาลง) ถ้าเก็บเป็น uint8 ค่าลบจะถูกตัดเป็น 0 ขอบหายไปครึ่งหนึ่ง ค่อยแปลงกลับด้วย convertScaleAbs ตอนท้าย

15.10 Contour — ขอบวัตถุที่เรียงต่อกัน

คอนทัวร์คือลำดับพิกเซลขอบที่เรียงรอบวัตถุ (อัลกอริทึม Moore เดินสำรวจเพื่อนบ้าน 8 ทิศ จนกลับมาที่จุดเริ่ม) เก็บเป็นลำดับทิศทางได้ จึงวัดรูปทรงได้โดยไม่ต้องเก็บภาพทั้งใบ

contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_NONE)
print("เจอวัตถุ", len(contours), "ชิ้น")

for c in contours:
    area = cv2.contourArea(c)          # พื้นที่
    if area < 100:                     # กรองเศษเล็ก ๆ ทิ้ง
        continue
    peri = cv2.arcLength(c, True)      # เส้นรอบรูป
    x, y, w, h = cv2.boundingRect(c)   # กรอบสี่เหลี่ยมที่ครอบพอดี
    cv2.rectangle(img, (x, y), (x + w, y + h), (0, 255, 0), 2)
    print(f"พื้นที่ {area:.0f} เส้นรอบรูป {peri:.0f} สัดส่วน {w/h:.2f}")

cv2.drawContours(img, contours, -1, (0, 0, 255), 1)   # -1 = วาดทุกเส้น
ค่าที่ใส่ความหมาย
RETR_EXTERNALเอาแค่ขอบนอกสุด ไม่เอารูข้างใน
RETR_TREEเอาทั้งหมดพร้อมลำดับชั้นว่าอันไหนอยู่ในอันไหน
CHAIN_APPROX_NONEเก็บทุกจุดขอบ
CHAIN_APPROX_SIMPLEเก็บแค่หัวมุม — เส้นตรงยาว ๆ เหลือ 2 จุด
findContours ต้องการภาพไบนารี — ใส่ภาพสีหรือภาพเทาเข้าไปตรง ๆ ไม่ได้ ลำดับที่ถูกคือ เทา → เบลอ → threshold → morphology → findContours

15.11 ส่วนเสริมที่ใช้บ่อย

Median filter — noise แบบ salt & pepper คือพิกเซลที่กระโดดไป 0 หรือ 255 แบบสุ่ม เกาส์เซียนจะเฉลี่ยค่าผิดเข้าไปด้วยจนเหลือรอยด่าง ส่วนมีเดียนเรียงค่าใน 9 ช่องแล้วหยิบตัวกลาง ค่าสุดโต่งจึงถูกตัดทิ้งหมดและขอบยังคม

med = cv2.medianBlur(img, 5)          # ขนาดต้องเป็นเลขคี่

Connected component labeling — นับว่ามีวัตถุกี่ชิ้นและแต่ละชิ้นใหญ่แค่ไหน

n, labels, stats, centroids = cv2.connectedComponentsWithStats(binary, connectivity=8)

for i in range(1, n):                 # เริ่มที่ 1 เพราะ 0 คือพื้นหลัง
    x, y, w, h, area = stats[i]
    if area < 50:                     # กรองก้อนเล็กออก
        continue
    cx, cy = centroids[i]
    print(f"วัตถุ {i}: พื้นที่ {area} จุดกึ่งกลาง ({cx:.0f}, {cy:.0f})")

Bit-plane slicing — ค่าพิกเซลหนึ่งไบต์แยกได้เป็น 8 ระนาบบิต บิตสูง (7, 6, 5) เก็บโครงร่างหลักของภาพ ส่วนบิตต่ำ (2, 1, 0) แทบเป็นสัญญาณรบกวน

for k in range(8):
    plane = ((gray >> k) & 1) * 255   # ดึงบิตที่ k ออกมาเป็นภาพขาวดำ
    cv2.imwrite(f"bit{k}.png", plane)
ความจริงข้อนี้คือหัวใจของการบีบอัดภาพ — ตัดบิตต่ำทิ้งแล้วภาพแทบไม่ต่าง และเป็นที่ซ่อนข้อมูลของ steganography ด้วย เพราะแก้บิตล่างสุดแล้วตาคนมองไม่ออก

15.12 ท่ามาตรฐาน — นับวัตถุในภาพ

รวมทุกหัวข้อในบทนี้เป็นลำดับเดียวที่ใช้จริงเกือบทุกงาน

import cv2
import numpy as np

img = cv2.imread("coins.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)          # 1) ลดเหลือช่องเดียว
blur = cv2.GaussianBlur(gray, (5, 5), 0)              # 2) ลด noise

T, binary = cv2.threshold(blur, 0, 255,               # 3) แยกวัตถุจากพื้นหลัง
                          cv2.THRESH_BINARY + cv2.THRESH_OTSU)

se = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (5, 5))
binary = cv2.morphologyEx(binary, cv2.MORPH_OPEN,  se)   # 4) เก็บกวาดเศษ
binary = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, se)   #    และอุดรู

contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL,  # 5) หาขอบวัตถุ
                               cv2.CHAIN_APPROX_SIMPLE)

count = 0
for c in contours:
    area = cv2.contourArea(c)
    if area < 300:                                    # 6) กรองก้อนเล็กทิ้ง
        continue
    count += 1
    x, y, w, h = cv2.boundingRect(c)
    cv2.rectangle(img, (x, y), (x + w, y + h), (0, 255, 0), 2)
    cv2.putText(img, str(count), (x, y - 5),
                cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 0, 255), 2)

print(f"Otsu เลือก T = {T:.0f} · นับได้ {count} ชิ้น")
cv2.imshow("count", img)
cv2.waitKey(0)
cv2.destroyAllWindows()
ตัวเลข 300 กับ (5,5) ต้องปรับตามภาพจริง — ไม่มีค่าที่ใช้ได้กับทุกรูป เขียนโค้ดให้แก้ตัวเลขพวกนี้ง่าย ๆ ไว้ตั้งแต่แรก แล้วนั่งลองปรับดูผล นี่คือ 80% ของงานจริง

15.13 Flowchart ของงานประมวลผลภาพ

งานประมวลผลภาพส่วนใหญ่เป็นท่อ (pipeline) — ภาพไหลผ่านขั้นตอนเรียงกันลงมา flowchart จึงเป็นเส้นตรงยาว แล้วมี loop ตอนท้ายเมื่อต้องจัดการวัตถุทีละชิ้น ในกล่องเขียนเป็นคำอธิบายสั้น ๆ ได้ ไม่ต้องใส่พารามิเตอร์ครบ

วาด pipeline ก่อนเขียนโค้ด ช่วยให้เห็นว่าขั้นไหนรับภาพชนิดไหน เช่น threshold ต้องได้ภาพ gray · findContours ต้องได้ภาพไบนารี ถ้าสลับลำดับผิดจะ error หรือได้ผลเพี้ยน

แบบทดสอบ · เลือกตอบ

เลือกได้ข้อละครั้งเดียว คิดก่อนกด

ภาพดิจิทัลจริง ๆ แล้วคืออะไร?
ทุกอย่างในบทนี้ทำได้เพราะภาพเป็นแค่ตัวเลข — บวก ลบ เทียบค่า วนลูปได้เหมือนข้อมูลอื่น
ลด L (จำนวนระดับความสว่าง) ให้เหลือน้อย ๆ ภาพจะเป็นยังไง?
ลองเลื่อนดูรูปในหัวข้อ 15.1 — ส่วนภาพที่แตกเป็นบล็อกนั่นคือลด N (sampling)
ภาพไบนารี (binary) มีค่าได้กี่แบบ?
ได้จากการเทียบค่าเทากับเกณฑ์ T เข้มกว่า T เป็นดำ ที่เหลือเป็นขาว
ฮิสโทแกรมบอกอะไรเรา?
มันนับอย่างเดียว ไม่สนใจว่าพิกเซลอยู่ตรงไหน จึงบอกได้แค่การกระจายความสว่าง
อยากคัดแยกวัตถุสีเขียวจากภาพที่มีเงาพาดผ่าน ควรใช้อะไร?
เงาทำให้ V (ความสว่าง) เปลี่ยน แต่ H (สีอะไร) แทบไม่ขยับ
ข้อดีของ Otsu คืออะไร?
มันไล่ลอง T ทุกค่า แล้วเลือกค่าที่แยกวัตถุกับพื้นหลังได้ขาดที่สุด
ภาพไบนารีมีจุดขาวเล็ก ๆ กระจายเป็น noise เต็มไปหมด ใช้อะไรลบ?
Opening กัดจุดเล็กหายไปก่อน แล้วค่อยพองวัตถุจริงกลับมาขนาดเดิม ส่วน Closing ใช้อุดรูในวัตถุ
ทำไมต้องเบลอภาพก่อนหาขอบ?
ขอบคือ “จุดที่ความสว่างเปลี่ยนเร็ว” — noise หนึ่งเม็ดก็เข้าเกณฑ์นั้นเหมือนกัน
ก่อนใช้ findContours ต้องเตรียมภาพเป็นแบบไหน?
ลำดับมาตรฐานคือ เทา → เบลอ → threshold → morphology → findContours
ข้อใดเรียงลำดับกล่องใน pipeline นับวัตถุ (15.13) ถูกต้อง?
threshold ต้องได้ภาพ gray · findContours ต้องได้ภาพไบนารี · ลำดับผิด = error หรือผลเพี้ยน

โจทย์ 1 — ดูค่าความสว่างของรูป

อ่านรูป แปลงเป็นเทา แล้วพิมพ์ค่าต่ำสุด สูงสุด และค่าเฉลี่ย จากนั้นนับว่ามีพิกเซลที่เข้มกว่า 128 อยู่กี่จุด

▸ ดูเฉลย 
import cv2
import numpy as np

img = cv2.imread("cat.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

print("ต่ำสุด :", gray.min())
print("สูงสุด :", gray.max())
print("เฉลี่ย :", round(gray.mean(), 1))

dark = int(np.sum(gray < 128))       # นับจุดที่เข้มกว่า 128
print(f"จุดเข้ม {dark} จุด จากทั้งหมด {gray.size} จุด")

โจทย์ 2 — เทียบฟิลเตอร์ 2 แบบ

อ่านรูป แล้วบันทึกออกมา 3 ไฟล์: ภาพเทาต้นฉบับ, ผ่าน GaussianBlur และผ่าน medianBlur แล้วเปิดดูเทียบกันว่าต่างกันตรงไหน

▸ ดูเฉลย 
import cv2

gray = cv2.imread("cat.jpg", cv2.IMREAD_GRAYSCALE)   # อ่านเป็นภาพเทาเลย
if gray is None:
    print("ไม่พบรูป")
    exit()

outputs = [
    ("out_gray.png",   gray),
    ("out_gauss.png",  cv2.GaussianBlur(gray, (5, 5), 0)),
    ("out_median.png", cv2.medianBlur(gray, 5)),
]

for name, image in outputs:
    cv2.imwrite(name, image)
    print("บันทึกแล้ว:", name)

โจทย์ 3 — นับวัตถุในภาพ

ทำตามลำดับมาตรฐาน: เทา → เบลอ → Otsu → findContours แล้วนับว่ามีวัตถุกี่ชิ้น โดยข้ามก้อนที่เล็กกว่า 200 พิกเซล และวาดกรอบสี่เหลี่ยมล้อมทุกชิ้นที่นับได้

▸ ดูเฉลย 
import cv2

img = cv2.imread("coins.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
blur = cv2.GaussianBlur(gray, (5, 5), 0)
_, binary = cv2.threshold(blur, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

contours, _ = cv2.findContours(binary, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

count = 0
for c in contours:
    if cv2.contourArea(c) < 200:      # ก้อนเล็กเกินไป ข้ามไป
        continue
    count += 1
    x, y, w, h = cv2.boundingRect(c)
    cv2.rectangle(img, (x, y), (x + w, y + h), (0, 255, 0), 2)

print("นับได้", count, "ชิ้น")
cv2.imwrite("out_count.jpg", img)

โจทย์ 4 — เมนูแปลงภาพ

ทำเมนูง่าย ๆ ด้วย while + if/elif ให้เลือกได้ว่าจะทำ 1) เทา 2) Otsu 3) หาขอบ แล้วบันทึกไฟล์ผลลัพธ์ โดยทุกครั้งทำจากภาพต้นฉบับ ไม่ใช่ภาพที่แปลงไปแล้ว

▸ ดูเฉลย 
import cv2

img = cv2.imread("cat.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

while True:
    print("1) เทา  2) Otsu  3) หาขอบ  q) ออก")
    c = input("เลือก: ").strip()

    if c == "q":
        break
    elif c == "1":
        out, name = gray, "out_gray.png"
    elif c == "2":
        _, out = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
        name = "out_otsu.png"
    elif c == "3":
        out = cv2.Canny(cv2.GaussianBlur(gray, (5, 5), 0), 100, 200)
        name = "out_edge.png"
    else:
        print("ไม่มีตัวเลือกนั้น")
        continue

    cv2.imwrite(name, out)
    print(f"บันทึก {name} · ความสว่างเฉลี่ย {out.mean():.1f}")
ทุกอย่างในบทนี้คือเลขคณิตกับลูป — OpenCV แค่ทำให้เร็วพอใช้งานจริง ถ้าเข้าใจว่าแต่ละคำสั่งทำอะไรกับตัวเลขในเมทริกซ์ ก็อ่านโค้ดงาน Computer Vision ที่ไหนก็รู้เรื่อง แม้จะเจอคำสั่งที่ไม่เคยเห็นมาก่อน

โจทย์ 5 — วาด Flowchart

วาด flowchart ของโปรแกรมหาขอบวัตถุ: อ่านภาพ → ขาวดำ → ลด noise → Canny → แสดงผล แล้วเขียนโค้ดตาม

img = cv2.imread("coins.jpg")
