6.1 รูปแบบพื้นฐาน

อ่านเป็นภาษาไทยว่า "สำหรับแต่ละตัวในชุดข้อมูล ให้ทำสิ่งนี้"

fruits = ["apple", "banana", "cherry"]

for f in fruits:
    print(f)

# apple
# banana
# cherry

Python จะหยิบสมาชิกทีละตัวใส่ตัวแปร f แล้วรันโค้ดที่ย่อหน้าข้างใน จนครบทุกตัวแล้วจึงออกจาก loop

ย่อหน้าคือทุกอย่าง: Python ใช้การเยื้อง (indent) บอกว่าโค้ดไหนอยู่ใน loop ปกติเยื้อง 4 ช่อง และห้ามลืมเครื่องหมาย : ท้ายบรรทัด for 
for f in fruits:
    print(f)          # อยู่ใน loop → รันทุกรอบ
print("จบแล้ว")        # อยู่นอก loop → รันครั้งเดียว

for i in range(5):

6.3 สะสมค่าใน loop

รูปแบบที่ใช้บ่อยที่สุดในโลก — เตรียมตัวแปรไว้ก่อน loop แล้วสะสมข้างใน

scores = [80, 75, 92, 68, 88]

total = 0                  # เตรียมไว้ "ก่อน" loop
for s in scores:
    total = total + s      # หรือเขียนสั้น ๆ total += s
print(total)               # 403

นับจำนวนตามเงื่อนไข ก็ใช้หลักเดียวกัน

count = 0
for s in scores:
    if s >= 80:
        count += 1
print(f"คะแนน 80 ขึ้นไปมี {count} คน")   # 3 คน

สร้าง list ใหม่จาก list เดิม

nums = [1, 2, 3, 4, 5]

squares = []
for n in nums:
    squares.append(n ** 2)
print(squares)      # [1, 4, 9, 16, 25]
เขียนสั้นได้ด้วย list comprehension:
squares = [n ** 2 for n in nums]
ให้ผลเหมือนกันเป๊ะ แต่ตอนเริ่มเรียนควรเขียนแบบ loop เต็ม ๆ ให้คล่องก่อน

6.4 enumerate() — เอาทั้ง index และค่า

เวลาต้องรู้ว่ากำลังอยู่ตัวที่เท่าไร คนใหม่มักเขียนแบบนี้

fruits = ["apple", "banana", "cherry"]

for i in range(len(fruits)):
    print(i, fruits[i])

ใช้ได้ แต่วิธีที่เป็น Python มากกว่าคือ

for i, f in enumerate(fruits):
    print(i, f)
# 0 apple
# 1 banana
# 2 cherry

# อยากให้เริ่มนับที่ 1
for i, f in enumerate(fruits, start=1):
    print(f"{i}. {f}")
# 1. apple
# 2. banana
# 3. cherry

6.5 zip() — วนสอง list พร้อมกัน

names  = ["ฟ้า", "นล", "อาร์ม"]
scores = [85, 92, 78]

for n, s in zip(names, scores):
    print(f"{n} ได้ {s} คะแนน")

# ฟ้า ได้ 85 คะแนน
# นล ได้ 92 คะแนน
# อาร์ม ได้ 78 คะแนน

zip จะหยุดที่ list ที่สั้นที่สุด ถ้า names มี 3 ตัว แต่ scores มี 2 ตัว จะวนแค่ 2 รอบ

6.7 break และ continue

# break = ออกจาก loop ทันที
for n in [3, 7, 12, 5, 9]:
    if n > 10:
        print("เจอเลขเกิน 10:", n)
        break
    print("ผ่าน:", n)
# ผ่าน: 3
# ผ่าน: 7
# เจอเลขเกิน 10: 12   ← หยุดตรงนี้ ไม่ไปต่อ

# continue = ข้ามรอบนี้ ไปรอบถัดไป
for n in range(1, 11):
    if n % 2 == 0:
        continue          # เลขคู่ข้ามไป
    print(n, end=" ")
# 1 3 5 7 9

ดูทีละบรรทัด · break

โค้ดเดียวกับข้างบน · ดูลูกศรตอนเจอเลขเกิน 10: กระโดดออกนอก loop ทันที 5 กับ 9 ไม่ถูกหยิบมาเลย

ดูทีละบรรทัด · continue

โค้ดเดียวกับข้างบน · ดูลูกศรตอนเจอเลขคู่: กลับไปหัว loop ทันที ข้าม print

6.8 Loop ซ้อน Loop

loop ข้างในจะวนครบทุกรอบ ต่อ 1 รอบของ loop ข้างนอก

for i in range(1, 4):
    for j in range(1, 4):
        print(f"{i}x{j}={i*j}", end="  ")
    print()          # ขึ้นบรรทัดใหม่ทุกจบแถว

# 1x1=1  1x2=2  1x3=3
# 2x1=2  2x2=4  2x3=6
# 3x1=3  3x2=6  3x3=9

ดูทีละบรรทัด · loop ซ้อน loop

โค้ดเดียวกับข้างบน · สังเกตว่า j วนครบ 1, 2, 3 ก่อน แล้ว i ถึงจะขยับ 1 ครั้ง และ j เริ่มนับใหม่ทุกรอบของ i

วาดรูปสามเหลี่ยมด้วย loop ซ้อน

for i in range(1, 6):
    for j in range(i):
        print("*", end="")
    print()

# *
# **
# ***
# ****
# *****

# วิธีสั้นกว่า ใช้การคูณ string
for i in range(1, 6):
    print("*" * i)
ระวัง: loop ซ้อน 2 ชั้นที่วนอย่างละ 1,000 รอบ = ทำงาน 1,000,000 รอบ ถ้าโปรแกรมช้าผิดปกติ ให้มองหา loop ซ้อนก่อนเป็นอันดับแรก
loop นอกวน 3 รอบ loop ในวน 2 รอบ คำสั่งใน loop ในทำงานกี่ครั้ง?
loop ในวนจนครบทุกครั้งที่ loop นอกขยับ 1 รอบ จึงเป็น 3 × 2 = 6

6.9 Flowchart ของ for loop

flowchart ไม่มีสัญลักษณ์ "for" โดยเฉพาะ loop คือเพชรที่ถามว่ายังมีค่าถัดไปไหม ถ้ามี (จริง) ทำงานในตัว loop แล้วลากเส้นย้อนกลับขึ้นไปถามใหม่ ถ้าหมดแล้ว (เท็จ) ออกไปทำบรรทัดถัดไป

n = int(input("n: "))
total = 0
for i in range(1, n + 1):
    total = total + i
print(total)
อ่าน flowchart ให้ออกว่าอะไรอยู่ใน loop: กล่องที่อยู่บนเส้นทางจากทาง "จริง" วนกลับไปหาเพชร = โค้ดที่ย่อหน้าอยู่ใต้ for ส่วนกล่องหลังทาง "เท็จ" = โค้ดที่อยู่นอก loop

Loop ซ้อน Loop — วงเล็กอยู่ในวงใหญ่

loop ซ้อนคือทั้งก้อน loop ใน (เพชร + เส้นวนกลับ) ไปอยู่บนทาง "จริง" ของ loop นอก ตัวอย่างสูตรคูณ 3x3 จาก 6.8 วาดได้แบบนี้

for i in range(1, 4):
    for j in range(1, 4):
        print(f"{i}x{j}={i*j}", end="  ")
    print()

ไล่ลูกศรตามทีละรอบ จะเห็นว่าวงเล็กวนครบก่อนวงใหญ่จะขยับ 1 ครั้งเสมอ

loop นอกloop ใน วนกี่รอบสิ่งที่ออกหน้าจอออกจาก loop ใน แล้ว
i = 1j = 1, 2, 31x1=1 1x2=2 1x3=3ขึ้นบรรทัดใหม่ · กลับไปถาม i
i = 2j = 1, 2, 32x1=2 2x2=4 2x3=6ขึ้นบรรทัดใหม่ · กลับไปถาม i
i = 3j = 1, 2, 33x1=3 3x2=6 3x3=9ขึ้นบรรทัดใหม่ · กลับไปถาม i
i หมดแล้ว——ออกทาง "เท็จ" ของ loop นอก → จบ
ดูจากเส้นว่ากล่องไหนอยู่ชั้นไหน
· อยู่ในกรอบฟ้า (วนตามเส้นซ้ายเล็ก) = ย่อหน้า 2 ชั้น ทำ 3 × 3 = 9 ครั้ง
· อยู่ในกรอบม่วงแต่นอกกรอบฟ้า เช่น print() = ย่อหน้า 1 ชั้น ทำ 3 ครั้ง
· อยู่นอกทุกกรอบ = ไม่ย่อหน้า ทำครั้งเดียว
ถ้าเอา print() ไปไว้ผิดชั้น ใน flowchart จะเห็นทันทีว่าอยู่ผิดวง

break และ continue — เส้นที่ลัดทาง

ใน flowchart สองคำสั่งนี้ไม่มีกล่องของตัวเอง แต่เป็นลูกศรที่ลัดทาง (เส้นแดง) break พุ่งออกนอก loop ไปเลย · continue ย้อนกลับไปหาเพชรของ loop โดยข้ามกล่องที่เหลือในรอบนั้น

for n in [3, 7, 12, 5, 9]:
    if n > 10:
        print("เจอเลขเกิน 10:", n)
        break
    print("ผ่าน:", n)
for n in range(1, 11):
    if n % 2 == 0:
        continue
    print(n, end=" ")
จำง่าย ๆ จากทิศของเส้นแดง: break ชี้ลงออกนอก loop · continue ชี้ขึ้นกลับไปหัว loop · ถ้าอยู่ใน loop ซ้อน ทั้งคู่มีผลแค่กับ loop ชั้นที่ใกล้ที่สุด (กรอบฟ้า) ไม่ทะลุออกกรอบม่วง

โจทย์ 2 — คัดกรองตัวเลข

กำหนด nums = [12, 7, 30, 5, 18, 9, 24] เขียน for loop เพื่อพิมพ์:

  1. เฉพาะเลขคู่
  2. ผลรวมของเลขที่มากกว่า 10
  3. จำนวนเลขคี่ทั้งหมด
▸ ดูเฉลย 
nums = [12, 7, 30, 5, 18, 9, 24]

print("เลขคู่:", end=" ")
for n in nums:
    if n % 2 == 0:
        print(n, end=" ")
print()

total = 0
odd = 0
for n in nums:
    if n > 10:
        total += n
    if n % 2 != 0:
        odd += 1

print("ผลรวมเลขที่มากกว่า 10:", total)   # 84
print("จำนวนเลขคี่:", odd)                # 3

โจทย์ 3 — ใบเสร็จ

มี items = ["ข้าวผัด", "ชาเย็น", "ส้มตำ"] และ prices = [50, 25, 45]
ใช้ zip() กับ enumerate() พิมพ์ใบเสร็จมีเลขลำดับเริ่มที่ 1 และบรรทัดสุดท้ายบอกราคารวม

▸ ดูเฉลย 
items  = ["ข้าวผัด", "ชาเย็น", "ส้มตำ"]
prices = [50, 25, 45]

total = 0
for i, (name, price) in enumerate(zip(items, prices), start=1):
    print(f"{i}. {name:10s} {price:4d} บาท")
    total += price

print("-" * 24)
print(f"   {'รวม':10s} {total:4d} บาท")

โจทย์ 4 — วาดรูปสามเหลี่ยม

ใช้ loop ซ้อน loop วาดรูปนี้ (รับความสูงจากผู้ใช้)

   ***
  *****
 *******

▸ ดูเฉลย 
h = int(input("ความสูง: "))

for i in range(1, h + 1):
    # ช่องว่างซ้าย ลดลงทีละ 1 ; ดาว เพิ่มทีละ 2
    for _ in range(h - i):
        print(" ", end="")
    for _ in range(2 * i - 1):
        print("*", end="")
    print()

# วิธีสั้นกว่า
for i in range(1, h + 1):
    print(" " * (h - i) + "*" * (2 * i - 1))

โจทย์ 5 — ท้าทาย

เขียนโปรแกรมหาจำนวนเฉพาะ (prime) ทั้งหมดตั้งแต่ 2 ถึง 50
ใบ้: จำนวนเฉพาะคือเลขที่หารลงตัวเฉพาะ 1 กับตัวเอง — ใช้ loop ซ้อนกับ break

▸ ดูเฉลย 
for n in range(2, 51):
    is_prime = True
    for d in range(2, n):          # ลองหารด้วย 2 ถึง n-1
        if n % d == 0:
            is_prime = False
            break                  # เจอตัวหารแล้ว ไม่ต้องหาต่อ
    if is_prime:
        print(n, end=" ")

โจทย์ 6 — วาด Flowchart

รับเลข n แล้วแสดงสูตรคูณแม่ n ตั้งแต่ n x 1 ถึง n x 12 วาด flowchart ก่อนเขียนโค้ด

▸ ดูเฉลย 
n = int(input("แม่: "))
for i in range(1, 13):
    print(f"{n} x {i} = {n * i}")

โจทย์ 7 — วาด Flowchart loop ซ้อน

วาด flowchart ของโปรแกรมวาดสามเหลี่ยมดาว 5 แถวใน 6.8 (แถวที่ i มีดาว i ดวง) แล้วบอกว่ากล่อง "แสดง *" ทำงานทั้งหมดกี่ครั้ง

▸ ดูเฉลย 
for i in range(1, 6):
    for j in range(i):
        print("*", end="")
    print()
