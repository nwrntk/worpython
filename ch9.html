Chapter 09

Dictionary — ดิกชันนารี

เก็บข้อมูลเป็นคู่ key → value ค้นด้วยชื่อ ไม่ใช่ตัวเลขตำแหน่ง

9.1 ทำไมต้องมี dict

สมมติเก็บคะแนนนักเรียน ถ้าใช้ list เราต้องจำว่าใครอยู่ตำแหน่งไหน

names  = ["ฟ้า", "นล", "อาร์ม"]
scores = [85, 92, 78]

# อยากรู้คะแนนของ "นล" ต้องหา index ก่อน แล้วค่อยไปเปิดอีก list
i = names.index("นล")
print(scores[i])       # 92

ด้วย dict เราค้นด้วยชื่อได้ตรง ๆ

scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78}

print(scores["นล"])    # 92

ใช้ปีกกา { } เขียนเป็น key: value คั่นด้วยจุลภาค

student = {
    "name": "ฟ้าใส",
    "age": 20,
    "gpa": 3.45,
    "subjects": ["Python", "Calculus"]     # value เป็นอะไรก็ได้
}

empty = {}      # dict ว่าง
key ต้องเป็นชนิดที่แก้ไม่ได้: ใช้ str, int, tuple ได้ แต่ใช้ list เป็น key ไม่ได้ (นี่คือเหตุผลหนึ่งที่ tuple มีอยู่)
ส่วน value เป็นอะไรก็ได้ รวมถึง list หรือ dict ซ้อนอีกที

9.2 อ่าน เพิ่ม แก้ ลบ

student = {"name": "ฟ้าใส", "age": 20}

# อ่าน
print(student["name"])       # ฟ้าใส

# เพิ่ม key ใหม่ — แค่กำหนดค่าให้ key ที่ยังไม่มี
student["gpa"] = 3.45
print(student)   # {'name': 'ฟ้าใส', 'age': 20, 'gpa': 3.45}

# แก้ค่าเดิม — เขียนแบบเดียวกันเป๊ะ
student["age"] = 21

# ลบ
del student["gpa"]
gpa = student.pop("age")     # ลบพร้อมเอาค่าออกมา
KeyError: เรียก key ที่ไม่มีจะพังทันที — student["email"] → KeyError: 'email'

9.3 .get() — อ่านแบบไม่พัง

student = {"name": "ฟ้าใส", "age": 20}

print(student.get("email"))              # None (ไม่ error)
print(student.get("email", "ไม่ระบุ"))    # ไม่ระบุ  ← ค่าสำรอง

# เช็กก่อนใช้ก็ได้
if "email" in student:
    print(student["email"])
else:
    print("ยังไม่มีอีเมล")

in กับ dict เช็กที่ key ไม่ใช่ value

scores = {"ฟ้า": 85, "นล": 92}

print("ฟ้า" in scores)          # True  → เช็ก key
print(85 in scores)             # False → 85 เป็น value ไม่ใช่ key
print(85 in scores.values())    # True  → ถ้าจะเช็ก value ต้องแบบนี้

ดูเป็นภาพ · dict ทำอะไรกับลิ้นชัก

โค้ดจาก 9.2 และ 9.3 · สังเกตว่าตอนอ่านค่า dict กระโดดไปที่ key ตรง ๆ ต่างจาก list ที่ต้องไล่หาทีละช่อง

9.4 วน loop บน dict

scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78}

# วนธรรมดา = ได้ key
for name in scores:
    print(name, scores[name])

# วิธีที่นิยมที่สุด: .items() ได้ทั้ง key และ value
for name, score in scores.items():
    print(f"{name} ได้ {score} คะแนน")

# เอาเฉพาะ key หรือเฉพาะ value
print(list(scores.keys()))      # ['ฟ้า', 'นล', 'อาร์ม']
print(list(scores.values()))    # [85, 92, 78]
print(sum(scores.values()))     # 255
print(max(scores.values()))     # 92

.items() คืนคู่ (key, value) เป็น tuple ทุกรอบ — ที่เขียน for name, score in ... ได้ก็เพราะเป็นการ unpack tuple แบบเดียวกับบทที่ 5

9.5 นับของด้วย dict

งานคลาสสิกที่ dict ทำได้ดีที่สุด — นับว่าอะไรปรากฏกี่ครั้ง

text = "banana"
count = {}

for ch in text:
    if ch in count:
        count[ch] += 1
    else:
        count[ch] = 1

print(count)     # {'b': 1, 'a': 3, 'n': 2}

# เขียนสั้นลงด้วย .get() ค่าสำรอง 0
count = {}
for ch in text:
    count[ch] = count.get(ch, 0) + 1

ดูทีละบรรทัด · นับตัวอักษร

โค้ดเดียวกับข้างบน · ดู count ค่อย ๆ โตขึ้น และสังเกตว่ารอบไหนไปทางสร้าง key ใหม่ (บรรทัด 8) รอบไหนไปทางบวกเพิ่ม (บรรทัด 6)

9.6 คำสั่งอื่นที่ใช้บ่อย

คำสั่งความหมาย
len(d)จำนวนคู่ key-value
d.keys()key ทั้งหมด
d.values()value ทั้งหมด
d.items()คู่ (key, value) ทั้งหมด
d.get(k, ค่าสำรอง)อ่านแบบไม่ error
d.pop(k)ลบ k พร้อมคืนค่า
d.update(d2)รวม d2 เข้า d (key ซ้ำจะทับของเดิม)
d.clear()ล้างทั้งหมด
dict(a=1, b=2)สร้าง dict อีกแบบ

9.7 dict ซ้อน dict

ข้อมูลจริงมักซับซ้อนกว่าคู่เดียว — ซ้อนกันได้เรื่อย ๆ

students = {
    "6510123": {"name": "ฟ้าใส", "gpa": 3.45},
    "6510124": {"name": "นล",    "gpa": 3.10},
}

print(students["6510123"]["name"])     # ฟ้าใส  → ไล่เข้าไปทีละชั้น

for sid, info in students.items():
    print(f"{sid} {info['name']} เกรด {info['gpa']}")
สังเกตเครื่องหมายคำพูดใน f-string: ข้างนอกใช้ " ข้างในต้องใช้ ' สลับกัน ไม่งั้น Python จะงงว่า string จบตรงไหน

list ของ dict — ตารางข้อมูลแบบที่ใช้งานจริง

ข้อมูลหลายรายการที่แต่ละรายการมีหลายฟิลด์ (เหมือนหนึ่งแถวในไฟล์ CSV บทที่ 13) นิยมเก็บเป็น list ของ dict

students = [
    {"name": "ฟ้า", "gpa": 3.45},
    {"name": "นล", "gpa": 2.80},
]

print(students[0]["name"])
students.append({"name": "อาร์ม", "gpa": 3.10})

for s in students:
    print(s["name"], s["gpa"])

best = max(students, key=lambda s: s["gpa"])
print("เก่งสุด:", best["name"])
# ฟ้า
# ฟ้า 3.45
# นล 2.8
# อาร์ม 3.1
# เก่งสุด: ฟ้า

อ่านจากนอกเข้าใน: students[0]["name"] คือ "รายการที่ 0 ก่อน" ได้ dict หนึ่งก้อน แล้วค่อยเอา key "name" ของก้อนนั้น · เพิ่มคนใหม่ด้วย append ทั้ง dict

9.8 เรียงลำดับ dict

scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78}

# เรียงตาม key
for name in sorted(scores):
    print(name, scores[name])

# เรียงตาม value จากมากไปน้อย
for name, score in sorted(scores.items(), key=lambda x: x[1], reverse=True):
    print(name, score)
# นล 92
# ฟ้า 85
# อาร์ม 78

# หา key ที่ value สูงสุด
print(max(scores, key=scores.get))    # นล

9.9 Flowchart นับของด้วย dict

ท่านับของ (9.5) มีเพชร 2 อัน: อันบนคือ loop ส่วนอันล่างถามว่า เคยเจอ key นี้หรือยัง — คำตอบต่างกันก็ทำคนละอย่าง

counts = {}
for w in words:
    if w in counts:
        counts[w] += 1
    else:
        counts[w] = 1
print(counts)
counts[w] = counts.get(w, 0) + 1 ทำงานเหมือน flowchart ข้างบนทุกอย่าง แค่ย่อเพชรกับ 2 กล่องให้เหลือบรรทัดเดียว
error ที่จะเริ่มเจอในบทนี้
· KeyError — ขอ key ที่ไม่มี หรือนับของโดยไม่สร้าง key ก่อน

แบบทดสอบ · เลือกตอบ

เลือกได้ข้อละครั้งเดียว คิดก่อนกด

d = {"a": 1, "b": 2} แล้ว d["c"] ได้อะไร?
อ่าน key ที่ไม่มีด้วย [ ] จะพัง ถ้าไม่อยากพังใช้ d.get("c") ซึ่งคืน None
อยากวนเอาทั้งชื่อและคะแนนพร้อมกัน เขียนยังไง?
.items() คืนคู่ (key, value) — วน dict เฉย ๆ จะได้แค่ key
scores = {"ฟ้า": 85} แล้ว 85 in scores ได้อะไร?
in เช็กที่ key เท่านั้น ถ้าจะเช็ก value ต้อง 85 in scores.values()
ข้อใดใช้เป็น key ของ dict ไม่ได้?
key ต้องเป็นชนิดที่แก้ไม่ได้ — list แก้ได้จึงใช้ไม่ได้ แต่ tuple ใช้ได้
โค้ดนี้ได้ผลอะไร?
d = {"a": 1}
d["a"] = 5
print(d)
key ห้ามซ้ำ — กำหนดค่าให้ key เดิมคือการแก้ค่า ไม่ใช่เพิ่มคู่ใหม่
d.get("x", 0) ต่างจาก d["x"] ยังไง?
พารามิเตอร์ตัวที่ 2 คือค่าสำรองเมื่อไม่เจอ key (ถ้าไม่ใส่จะได้ None)
ใน flowchart นับคำ (9.9) ถ้าตัดเพชร w in counts ? ทิ้ง แล้วทำ counts[w] += 1 ทุกรอบ จะเกิดอะไร?
คำแรกยังไม่มี key ใน dict การ += 1 ต้องอ่านค่าเดิมก่อน จึงพัง · ทาง "เท็จ" มีไว้สร้าง key นี่เอง

แบบทดสอบ · เติมช่องว่าง

เติมช่องว่าง · dict: อ่าน เพิ่ม และวนดูทั้งหมด

key ที่ไม่มีอยู่ ต้องใช้วิธีที่ไม่ทำให้โปรแกรมพัง

แบบทดสอบ · เขียนโค้ด

โจทย์ 1 — สมุดโทรศัพท์

สร้าง dict เก็บชื่อกับเบอร์โทร 3 คน แล้ว: เพิ่มคนที่ 4, แก้เบอร์ของคนแรก, ลบคนที่ 2, และพิมพ์ทุกคู่ด้วย .items()

▸ ดูเฉลย 
phone = {"ฟ้า": "081-1111111", "นล": "082-2222222", "อาร์ม": "083-3333333"}

phone["มิน"] = "084-4444444"
phone["ฟ้า"] = "089-9999999"
del phone["นล"]

for name, num in phone.items():
    print(f"{name:6s} {num}")

โจทย์ 2 — นับตัวอักษร

รับข้อความจากผู้ใช้ แล้วนับว่าตัวอักษรแต่ละตัวปรากฏกี่ครั้ง (ไม่ต้องนับช่องว่าง) พิมพ์ผลเรียงจากตัวที่เจอบ่อยที่สุด

▸ ดูเฉลย 
text = input("ข้อความ: ")

count = {}
for ch in text:
    if ch == " ":
        continue
    count[ch] = count.get(ch, 0) + 1

for ch, n in sorted(count.items(), key=lambda x: x[1], reverse=True):
    print(f"{ch} : {n}")

โจทย์ 3 — สรุปคะแนน

กำหนด scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78, "มิน": 95, "เก้า": 66}
พิมพ์: จำนวนคน, คะแนนเฉลี่ย, ชื่อคนที่ได้คะแนนสูงสุด, และรายชื่อคนที่ได้ต่ำกว่าค่าเฉลี่ย

▸ ดูเฉลย 
scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78, "มิน": 95, "เก้า": 66}

avg = sum(scores.values()) / len(scores)

print("จำนวนคน:", len(scores))
print("เฉลี่ย:", round(avg, 2))
print("สูงสุด:", max(scores, key=scores.get))

print("ต่ำกว่าเฉลี่ย:")
for name, s in scores.items():
    if s < avg:
        print(" -", name, s)

โจทย์ 4 — ท้าทาย: ระบบสต๊อกสินค้า

ทำโปรแกรมเมนูวนซ้ำ (ใช้ while + dict ซ้อน dict):
1) เพิ่มสินค้า (ชื่อ, ราคา, จำนวน)   2) ขายสินค้า (ลดจำนวน)   3) แสดงสต๊อกทั้งหมดพร้อมมูลค่ารวม   4) ออก

▸ ดูเฉลย 
stock = {}

while True:
    print("\n1) เพิ่ม  2) ขาย  3) แสดง  4) ออก")
    c = input("เลือก: ")

    if c == "1":
        name = input("ชื่อสินค้า: ")
        stock[name] = {
            "price": float(input("ราคา: ")),
            "qty":   int(input("จำนวน: ")),
        }
        print("เพิ่มแล้ว")

    elif c == "2":
        name = input("ขายสินค้า: ")
        if name not in stock:
            print("ไม่มีสินค้านี้")
        else:
            n = int(input("ขายกี่ชิ้น: "))
            if n > stock[name]["qty"]:
                print("ของไม่พอ เหลือ", stock[name]["qty"])
            else:
                stock[name]["qty"] -= n
                print("ขายแล้ว เหลือ", stock[name]["qty"])

    elif c == "3":
        if not stock:
            print("ยังไม่มีสินค้า")
        total = 0
        for name, info in stock.items():
            value = info["price"] * info["qty"]
            total += value
            print(f"{name:12s} {info['price']:8.2f} x {info['qty']:3d} = {value:9.2f}")
        print(f"{'มูลค่ารวม':12s} {total:24.2f}")

    elif c == "4":
        break

    else:
        print("เลือก 1-4 เท่านั้น")
