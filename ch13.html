13.1 อ่านไฟล์

ใช้ with open(...) เสมอ — Python จะปิดไฟล์ให้อัตโนมัติเมื่อจบบล็อก แม้โปรแกรมจะ error กลางทาง

with open("data.txt", "r", encoding="utf-8") as f:
    content = f.read()          # อ่านทั้งไฟล์เป็น string เดียว

print(content)
ภาษาไทยต้องใส่ encoding="utf-8" เสมอ
ถ้าไม่ใส่ Windows จะเดา encoding เอง แล้วภาษาไทยจะกลายเป็นตัวประหลาด หรือเกิด UnicodeDecodeError

3 วิธีอ่าน เลือกตามงาน

# 1) ทั้งไฟล์เป็น string เดียว
with open("data.txt", encoding="utf-8") as f:
    text = f.read()

# 2) เป็น list ทีละบรรทัด (ติด \n ท้ายมาด้วย)
with open("data.txt", encoding="utf-8") as f:
    lines = f.readlines()

# 3) วนทีละบรรทัด — ประหยัดหน่วยความจำที่สุด ใช้บ่อยที่สุด
with open("data.txt", encoding="utf-8") as f:
    for line in f:
        print(line.strip())     # strip() ตัด \n ท้ายบรรทัดทิ้ง

13.2 เขียนไฟล์

โหมดความหมาย
"r"อ่าน (ค่าเริ่มต้น) — ถ้าไม่มีไฟล์จะ error
"w"เขียนใหม่ — ลบข้อมูลเดิมทั้งหมด
"a"เขียนต่อท้าย — ของเดิมยังอยู่
"x"สร้างใหม่ — ถ้ามีไฟล์อยู่แล้วจะ error
# เขียนทับ
with open("output.txt", "w", encoding="utf-8") as f:
    f.write("บรรทัดแรก\n")           # ต้องใส่ \n เอง
    f.write("บรรทัดที่สอง\n")

# เขียนต่อท้าย
with open("output.txt", "a", encoding="utf-8") as f:
    f.write("บรรทัดที่สาม\n")

# เขียน list ทีเดียว
names = ["ฟ้า", "นล", "อาร์ม"]
with open("names.txt", "w", encoding="utf-8") as f:
    for n in names:
        f.write(n + "\n")
โหมด "w" ลบของเดิมทันทีที่เปิดไฟล์ — ก่อนที่จะเขียนอะไรด้วยซ้ำ ถ้าไม่ได้ตั้งใจลบ ให้ใช้ "a"

13.3 อ่าน-เขียนข้อมูลแบบตาราง

# เขียน: ชื่อ,คะแนน บรรทัดละคน
scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78}

with open("scores.csv", "w", encoding="utf-8") as f:
    for name, s in scores.items():
        f.write(f"{name},{s}\n")

# อ่านกลับเข้ามาเป็น dict
loaded = {}
with open("scores.csv", encoding="utf-8") as f:
    for line in f:
        line = line.strip()
        if not line:                 # ข้ามบรรทัดว่าง
            continue
        name, s = line.split(",")
        loaded[name] = int(s)        # ที่อ่านมาเป็น str ต้องแปลง

print(loaded)      # {'ฟ้า': 85, 'นล': 92, 'อาร์ม': 78}
โค้ดนี้เขียนทับ scores.csv ด้วยโหมด "w" ไฟล์ตัวอย่าง 6 คนจะเหลือ 3 คน อยากได้ของเดิมคืนให้ดาวน์โหลดใหม่
ข้อมูลจากไฟล์เป็น string เสมอ — เลข 85 ที่อ่านมาคือ "85" ต้อง int() หรือ float() ก่อนเอาไปคำนวณ

13.4 try-except — กันโปรแกรมพัง

try:
    n = int(input("ใส่ตัวเลข: "))
    print(100 / n)
except ValueError:
    print("นั่นไม่ใช่ตัวเลข")
except ZeroDivisionError:
    print("หารด้วยศูนย์ไม่ได้")

โค้ดใน try ถ้าพังตรงไหน จะกระโดดไปหา except ที่ตรงกับชนิดของ error ทันที ส่วนที่เหลือใน try ไม่ทำงาน

Exceptionเกิดเมื่อ
ValueErrorint("abc") — ชนิดถูกแต่ค่าใช้ไม่ได้
TypeError"a" + 1 — ชนิดข้อมูลผิด
ZeroDivisionErrorหารด้วย 0
IndexErrora[99] — index เกินขอบ
KeyErrord["ไม่มี"] — key ไม่มีใน dict
FileNotFoundErrorเปิดไฟล์ที่ไม่มีอยู่

ดูทีละบรรทัด · try-except

แทน input() ด้วย list ของสิ่งที่ผู้ใช้อาจพิมพ์ 3 แบบ · ดูว่าพอเกิด error บรรทัดที่เหลือใน try ถูกข้ามทันที แล้วกระโดดไปหา except ที่ชนิดตรงกัน

13.5 โครงสร้างเต็ม

try:
    f = open("data.txt", encoding="utf-8")
    data = f.read()
except FileNotFoundError:
    print("ไม่พบไฟล์")
else:
    print("อ่านสำเร็จ", len(data), "ตัวอักษร")   # ทำเมื่อไม่มี error
finally:
    print("จบการทำงาน")                          # ทำทุกกรณี
ส่วนทำงานเมื่อ
tryเสมอ — เป็นโค้ดที่เสี่ยงพัง
exceptเกิด error ชนิดที่ระบุ
elseไม่เกิด error เลย
finallyทุกกรณี ทั้งพังและไม่พัง

ดักหลายชนิดพร้อมกัน และดูรายละเอียด error ได้

try:
    n = int(input("ตัวเลข: "))
    print(10 / n)
except (ValueError, ZeroDivisionError) as e:
    print("ผิดพลาด:", e)
อย่าเขียน except: เปล่า ๆ เพราะมันกลืน error ทุกชนิด รวมถึง bug ที่เราเขียนผิดเอง ทำให้หาปัญหาไม่เจอ — ระบุชนิดที่คาดไว้เสมอ

13.6 รูปแบบที่ใช้จริง

def read_scores(filename):
    """อ่านไฟล์คะแนน คืน dict — ถ้าไม่มีไฟล์คืน dict ว่าง"""
    scores = {}
    try:
        with open(filename, encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                name, s = line.split(",")
                scores[name] = int(s)
    except FileNotFoundError:
        print(f"ยังไม่มีไฟล์ {filename} เริ่มจากข้อมูลว่าง")
    except ValueError:
        print("รูปแบบไฟล์ผิด ข้ามข้อมูลที่เสีย")
    return scores


def save_scores(filename, scores):
    with open(filename, "w", encoding="utf-8") as f:
        for name, s in scores.items():
            f.write(f"{name},{s}\n")


scores = read_scores("scores.csv")
scores["มิน"] = 95
save_scores("scores.csv", scores)
ลองกับ scores_bad.csv · read_scores("scores_bad.csv") ได้แค่ {'ฟ้า': 85, 'นล': 92} เพราะ try ครอบทั้ง loop พอเจอบรรทัดเสียบรรทัดแรก ก็กระโดดออกไป except ทันที คนที่เหลือไม่ถูกอ่านเลย · เทียบกับเฉลยโจทย์ 3 ที่วาง try ไว้ใน loop ทีละบรรทัด จะข้ามแค่บรรทัดเสียและได้ครบ 4 คน · ตำแหน่งของ try กำหนดว่าพังแล้วเสียหายแค่ไหน

รับ input แบบวนจนกว่าจะถูก — รวม while + try

while True:
    try:
        age = int(input("อายุ: "))
        break
    except ValueError:
        print("ใส่เป็นตัวเลขเท่านั้น")

print("อายุของคุณคือ", age)

13.7 Flowchart ของ try-except

try-except วาดเป็นเพชรที่ถามว่า "เกิด error นี้ไหม?" ต่อจากกล่องที่เสี่ยงพัง ทาง "จริง" คือบล็อก except ทาง "เท็จ" คือทำต่อตามปกติ

try:
    f = open("data.txt", encoding="utf-8")
except FileNotFoundError:
    print("ไม่พบไฟล์")
else:
    total = 0
    for line in f:
        total += int(line)
    f.close()
    print(total)
error ที่จะเริ่มเจอในบทนี้
· FileNotFoundError — หาไฟล์ไม่เจอ
· ValueError — แปลงค่าที่ผู้ใช้พิมพ์ผิดไม่ได้
ดูทุกชนิดได้ที่ คู่มืออ่าน error

แบบทดสอบ · เลือกตอบ

เลือกได้ข้อละครั้งเดียว คิดก่อนกด

เปิดไฟล์ด้วยโหมด "w" ทั้งที่ไฟล์นั้นมีข้อมูลอยู่แล้ว จะเกิดอะไรขึ้น?
"w" ล้างไฟล์ทันทีที่เปิด ถ้าจะเขียนต่อท้ายต้องใช้ "a"
ทำไมถึงควรใช้ with open(...) แทน open(...) เฉย ๆ?
ไฟล์ที่ไม่ปิดอาจทำให้ข้อมูลที่เขียนยังไม่ถูกบันทึกลงดิสก์จริง
int("สวัสดี") ทำให้เกิด exception ชนิดใด?
ชนิดข้อมูล (str) ถูกต้อง แต่ค่าข้างในแปลงเป็นเลขไม่ได้ → ValueError
finally ทำงานเมื่อไร?
ใช้เก็บงานปิดท้ายที่ต้องทำแน่ ๆ — ส่วน else ต่างหากที่ทำเฉพาะตอนไม่เกิด error
อ่านตัวเลขจากไฟล์มาได้ line = "85" แล้วทำ line + 5 จะเกิดอะไรขึ้น?
ข้อมูลจากไฟล์เป็น str เสมอ — ต่อ str กับ int ไม่ได้ ต้อง int(line) + 5
ทำไมไม่ควรเขียน except: เปล่า ๆ?
ระบุชนิดที่คาดไว้เสมอ เพื่อให้ error ที่ไม่ได้คาดยังแสดงออกมาให้เห็น
ไฟล์ภาษาไทยควรเปิดยังไง?
ไม่ระบุ encoding แล้ว Windows จะเดาเอง ทำให้ภาษาไทยเพี้ยนหรือ error
ใน flowchart ของ try-except ทาง "จริง" ของเพชร "เกิด error ?" ตรงกับโค้ดส่วนไหน?
เกิด error → ไปทำบล็อก except · ไม่เกิด (เท็จ) → ทำต่อตามปกติ ซึ่งก็คือส่วน else

▸ ดูเฉลย 
scores = {}

try:
    with open("scores.csv", encoding="utf-8") as f:

โจทย์ 4 — ท้าทาย: สมุดบันทึกที่ไม่หาย

ทำโปรแกรมจดบันทึกที่ข้อมูลอยู่ครบแม้ปิดโปรแกรมไปแล้ว — รวมทุกอย่างที่เรียนมา 13 บท:

  1. ตอนเปิดโปรแกรม อ่านบันทึกเดิมจาก notes.txt (ถ้ายังไม่มีไฟล์ให้เริ่มจากว่าง · ลองทั้งแบบไม่มีไฟล์ และแบบใช้ไฟล์ตัวอย่าง)
  2. เมนู: 1) เพิ่มบันทึก 2) ลบตามเลขลำดับ 3) ค้นหาคำ 4) แสดงทั้งหมด 5) ออก
  3. ทุกครั้งที่แก้ไข ให้บันทึกลงไฟล์ทันที
▸ ดูเฉลย 
FILENAME = "notes.txt"


def load():
    try:
        with open(FILENAME, encoding="utf-8") as f:
            return [line.strip() for line in f if line.strip()]
    except FileNotFoundError:
        return []


def save(notes):
    with open(FILENAME, "w", encoding="utf-8") as f:
        for n in notes:
            f.write(n + "\n")


def show(notes):
    if not notes:
        print("ยังไม่มีบันทึก")
        return
    for i, n in enumerate(notes, start=1):
        print(f"{i}. {n}")


notes = load()
print(f"โหลดบันทึกเดิม {len(notes)} รายการ")

while True:
    print("\n1) เพิ่ม  2) ลบ  3) ค้นหา  4) แสดง  5) ออก")
    c = input("เลือก: ").strip()

    if c == "1":
        notes.append(input("บันทึก: ").strip())
        save(notes)
        print("บันทึกแล้ว")

    elif c == "2":
        show(notes)
        try:
            i = int(input("ลบลำดับที่: "))
            removed = notes.pop(i - 1)      # ผู้ใช้นับจาก 1
            save(notes)
            print("ลบแล้ว:", removed)
        except ValueError:
            print("ใส่เป็นตัวเลข")
        except IndexError:
            print("ไม่มีลำดับนั้น")

    elif c == "3":
        word = input("ค้นหาคำ: ").lower()
        found = [n for n in notes if word in n.lower()]
        if found:
            show(found)
        else:
            print("ไม่พบ")

    elif c == "4":
        show(notes)

    elif c == "5":
        print("บันทึกไว้เรียบร้อย บ๊ายบาย")
        break

    else:
        print("เลือก 1-5 เท่านั้น")
