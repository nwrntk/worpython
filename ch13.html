13.1 เปิดไฟล์ แล้วต้องปิดด้วย

การใช้ไฟล์มี 3 ขั้นเสมอ: เปิด → ใช้งาน → ปิด · open() คืนตัวแทนไฟล์ (file object) ออกมาให้เราสั่งงานต่อ

f = open("data.txt", encoding="utf-8")   # 1) เปิด
text = f.read()                          # 2) ใช้งาน
f.close()                                # 3) ปิด

print(type(f))                                          # <class '_io.TextIOWrapper'>
print(f.name)                                           # data.txt
print(f.closed)                                         # True
print(len(text))                                        # 13
ถ้าลืม close() ไฟล์จะยังถูกจองไว้ · ที่เจอบ่อยคือสั่งเขียนแล้วไปเปิดอ่านทันที แต่ข้อมูลยังค้างอยู่ในหน่วยความจำ ยังไม่ลงไฟล์จริง เลยอ่านได้ไม่ครบ 
f = open("note.txt", "w", encoding="utf-8")
f.write("สวัสดี")
# ยังไม่ได้ close

print(repr(open("note.txt", encoding="utf-8").read()))   # ยังไม่ปิด    # ''  ← ยังไม่มีอะไรในไฟล์
f.close()
print(repr(open("note.txt", encoding="utf-8").read()))   # ปิดแล้ว      # 'สวัสดี'  ← ข้อมูลลงไฟล์แล้ว

with — ให้ Python ปิดไฟล์ให้เอง

เขียนแบบ with แล้วไม่ต้องเรียก close() เลย · พอจบบล็อก (ย่อหน้ากลับออกมา) Python ปิดให้ทันที แม้โค้ดจะ error กลางทาง ซึ่งเป็นจุดที่โค้ดแบบ open/close พลาดบ่อย เพราะพังก่อนถึงบรรทัด close()

with open("data.txt", encoding="utf-8") as f:
    text = f.read()
    print(f.closed)        # ข้างในบล็อก: ยังเปิดอยู่            # False

print(f.closed)            # ออกจากบล็อก: ปิดให้แล้วอัตโนมัติ    # True

เปิดหลายไฟล์พร้อมกันก็ได้ คั่นด้วยจุลภาค

with open("in.txt", encoding="utf-8") as src, open("out.txt", "w", encoding="utf-8") as dst:
    dst.write(src.read().upper())
สรุปกฎของบทนี้: ใช้ with open(...) เสมอ · เขียน open/close เองเฉพาะตอนที่ ต้องเปิดไฟล์ค้างไว้ข้ามฟังก์ชันจริง ๆ ซึ่งเจอไม่บ่อย
เปิดไฟล์ด้วย with open(...) as f: แล้วโค้ดข้างในเกิด error · ไฟล์ถูกปิดไหม?
นี่คือเหตุผลหลักที่ใช้ with · ถ้าเขียน open/close เอง แล้วพังกลางทาง บรรทัด close() จะไม่ถูกทำเลย

13.2 โหมดเปิดไฟล์

พารามิเตอร์ตัวที่สองของ open() คือโหมด บอกว่าจะทำอะไรกับไฟล์ · ไม่ใส่ = "r" คืออ่านอย่างเดียว

โหมดไฟล์ไม่มีอยู่ไฟล์มีอยู่แล้วใช้ตอนไหน
"r"FileNotFoundErrorอ่านได้ เขียนไม่ได้อ่านข้อมูล (ค่าเริ่มต้น)
"w"สร้างใหม่ให้ล้างข้อมูลเดิมทิ้งทั้งหมดเขียนผลลัพธ์ชุดใหม่
"a"สร้างใหม่ให้เขียนต่อท้าย ของเดิมอยู่ครบบันทึกเพิ่มทีละรายการ
"x"สร้างใหม่ให้FileExistsErrorกันเขียนทับโดยไม่ตั้งใจ
# "w" ล้างไฟล์ทิ้งตั้งแต่ตอนเปิด ยังไม่ทันเขียนอะไรเลย
with open("notes.txt", encoding="utf-8") as f:
    print(len(f.read()))      # ของเดิมมีข้อมูลอยู่                      # 70

f = open("notes.txt", "w", encoding="utf-8")
f.close()                     # เปิดโหมด w แล้วปิดเลย ไม่ได้เขียนอะไร

with open("notes.txt", encoding="utf-8") as f:
    print(len(f.read()))      # ของเดิมหายหมดแล้ว                        # 0
พิมพ์ "w" แทน "r" = ข้อมูลหาย และกู้คืนไม่ได้ · ถ้าแค่อ่าน ไม่ต้องใส่โหมดเลยปลอดภัยที่สุด · ถ้าไม่อยากเขียนทับไฟล์ที่มีอยู่ ให้ใช้ "x"

เติม "b" ท้ายโหมด แปลว่าอ่าน-เขียนเป็นไบต์ดิบ ไม่ใช่ข้อความ ใช้กับไฟล์ที่ไม่ใช่ตัวหนังสือ เช่น รูปภาพ (บทที่ 17–18) · โหมดไบต์ห้ามใส่ encoding

with open("cat.jpg", "rb") as f:     # rb = read binary
    head = f.read(4)

13.4 ตัวชี้ตำแหน่งในไฟล์

ไฟล์ที่เปิดอยู่มีตัวชี้บอกว่าอ่านถึงไหนแล้ว · อ่านไปข้างหน้าเรื่อย ๆ พออ่านจนจบไฟล์ ตัวชี้อยู่ท้ายสุด อ่านซ้ำอีกครั้งจึงได้ข้อความว่าง ซึ่งเป็นกับดักที่ทำคนงงมาก

with open("data.txt", encoding="utf-8") as f:
    first = f.read()
    second = f.read()          # อ่านซ้ำทันที

print(len(first))                                                    # 13
print(repr(second))            # ว่าง เพราะตัวชี้อยู่ท้ายไฟล์แล้ว    # ''

f.tell() ถามว่าตัวชี้อยู่ตำแหน่งไหน · f.seek(0) สั่งให้กลับไปต้นไฟล์

with open("data.txt", encoding="utf-8") as f:
    print(f.tell())            # เพิ่งเปิด อยู่ต้นไฟล์    # 0
    print(f.readline().strip())                           # 12
    print(f.tell())            # ขยับไปแล้ว               # 3
    f.seek(0)                  # ย้อนกลับไปต้นไฟล์
    print(f.readline().strip())                           # 12
วิธีที่ง่ายกว่าการ seek — ถ้าต้องใช้ข้อมูลหลายรอบ ให้อ่านเก็บใส่ตัวแปรครั้งเดียว เช่น lines = f.readlines() แล้ววน lines กี่รอบก็ได้ ไม่ต้องยุ่งกับตัวชี้
อ่านไฟล์ด้วย f.read() ไปแล้วหนึ่งครั้ง · สั่ง f.read() ซ้ำทันทีได้อะไร?
ตัวชี้เดินไปข้างหน้าอย่างเดียว · ถ้าต้องอ่านซ้ำให้ f.seek(0) ก่อน หรือเก็บไว้ในตัวแปรตั้งแต่รอบแรก

13.7 อ่าน-เขียนข้อมูลแบบตาราง

# เขียน: ชื่อ,คะแนน บรรทัดละคน
scores = {"ฟ้า": 85, "นล": 92, "อาร์ม": 78}

with open("scores.csv", "w", encoding="utf-8") as f:
    for name, s in scores.items():
        f.write(f"{name},{s}\n")

# อ่านกลับเข้ามาเป็น dict
loaded = {}
with open("scores.csv", encoding="utf-8") as f:
    for line in f:
        line = line.strip()
        if not line:                 # ข้ามบรรทัดว่าง
            continue
        name, s = line.split(",")
        loaded[name] = int(s)        # ที่อ่านมาเป็น str ต้องแปลง

print(loaded)      # {'ฟ้า': 85, 'นล': 92, 'อาร์ม': 78}
โค้ดนี้เขียนทับ scores.csv ด้วยโหมด "w" ไฟล์ตัวอย่าง 6 คนจะเหลือ 3 คน อยากได้ของเดิมคืนให้ดาวน์โหลดใหม่
ข้อมูลจากไฟล์เป็น string เสมอ — เลข 85 ที่อ่านมาคือ "85" ต้อง int() หรือ float() ก่อนเอาไปคำนวณ

13.8 try-except — กันโปรแกรมพัง

try:
    n = int(input("ใส่ตัวเลข: "))
    print(100 / n)
except ValueError:
    print("นั่นไม่ใช่ตัวเลข")
except ZeroDivisionError:
    print("หารด้วยศูนย์ไม่ได้")

โค้ดใน try ถ้าพังตรงไหน จะกระโดดไปหา except ที่ตรงกับชนิดของ error ทันที ส่วนที่เหลือใน try ไม่ทำงาน

พอเกิด error Python ทิ้งบรรทัดที่เหลือใน try ทันที แล้วไปหา except ที่ตรงชนิด · ส่วน finally ทำเสมอ

13.9 โครงสร้างเต็ม

try:
    f = open("data.txt", encoding="utf-8")
    data = f.read()
except FileNotFoundError:
    print("ไม่พบไฟล์")
else:
    print("อ่านสำเร็จ", len(data), "ตัวอักษร")   # ทำเมื่อไม่มี error
finally:
    print("จบการทำงาน")                          # ทำทุกกรณี
ส่วนทำงานเมื่อ
tryเสมอ — เป็นโค้ดที่เสี่ยงพัง
exceptเกิด error ชนิดที่ระบุ
elseไม่เกิด error เลย
finallyทุกกรณี ทั้งพังและไม่พัง

ดักหลายชนิดพร้อมกัน และดูรายละเอียด error ได้

try:
    n = int(input("ตัวเลข: "))

except (ValueError, ZeroDivisionError) as e:
    print("ผิดพลาด:", e)
อย่าเขียน except: เปล่า ๆ เพราะมันกลืน error ทุกชนิด รวมถึง bug ที่เราเขียนผิดเอง ทำให้หาปัญหาไม่เจอ — ระบุชนิดที่คาดไว้เสมอ

13.10 รูปแบบที่ใช้จริง

def read_scores(filename):
    """อ่านไฟล์คะแนน คืน dict — ถ้าไม่มีไฟล์คืน dict ว่าง"""
    scores = {}
    try:
        with open(filename, encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                name, s = line.split(",")
                scores[name] = int(s)
    except FileNotFoundError:
        print(f"ยังไม่มีไฟล์ {filename} เริ่มจากข้อมูลว่าง")
    except ValueError:
        print("รูปแบบไฟล์ผิด ข้ามข้อมูลที่เสีย")
    return scores


def save_scores(filename, scores):
    with open(filename, "w", encoding="utf-8") as f:
        for name, s in scores.items():
            f.write(f"{name},{s}\n")


scores = read_scores("scores.csv")
scores["มิน"] = 95
save_scores("scores.csv", scores)
ลองกับ scores_bad.csv · read_scores("scores_bad.csv") ได้แค่ {'ฟ้า': 85, 'นล': 92} เพราะ try ครอบทั้ง loop พอเจอบรรทัดเสียบรรทัดแรก ก็กระโดดออกไป except ทันที คนที่เหลือไม่ถูกอ่านเลย · เทียบกับเฉลยโจทย์ 3 ที่วาง try ไว้ใน loop ทีละบรรทัด จะข้ามแค่บรรทัดเสียและได้ครบ 4 คน · ตำแหน่งของ try กำหนดว่าพังแล้วเสียหายแค่ไหน

13.11 Flowchart ของ try-except

try-except วาดเป็นเพชรที่ถามว่า "เกิด error นี้ไหม?" ต่อจากกล่องที่เสี่ยงพัง ทาง "จริง" คือบล็อก except ทาง "เท็จ" คือทำต่อตามปกติ

try:
    f = open("data.txt", encoding="utf-8")
except FileNotFoundError:
    print("ไม่พบไฟล์")
else:
    total = 0
    for line in f:
        total += int(line)
    f.close()
    print(total)
error ที่จะเริ่มเจอในบทนี้
· FileNotFoundError — หาไฟล์ไม่เจอ
· ValueError — แปลงค่าที่ผู้ใช้พิมพ์ผิดไม่ได้
ดูทุกชนิดได้ที่ คู่มืออ่าน error

แบบทดสอบ · เลือกตอบ

เลือกได้ข้อละครั้งเดียว คิดก่อนกด

เปิดไฟล์ด้วยโหมด "w" ทั้งที่ไฟล์นั้นมีข้อมูลอยู่แล้ว จะเกิดอะไรขึ้น?
"w" ล้างไฟล์ทันทีที่เปิด ถ้าจะเขียนต่อท้ายต้องใช้ "a"
ทำไมถึงควรใช้ with open(...) แทน open(...) เฉย ๆ?
ไฟล์ที่ไม่ปิดอาจทำให้ข้อมูลที่เขียนยังไม่ถูกบันทึกลงดิสก์จริง
int("สวัสดี") ทำให้เกิด exception ชนิดใด?
ชนิดข้อมูล (str) ถูกต้อง แต่ค่าข้างในแปลงเป็นเลขไม่ได้ → ValueError
finally ทำงานเมื่อไร?
ใช้เก็บงานปิดท้ายที่ต้องทำแน่ ๆ — ส่วน else ต่างหากที่ทำเฉพาะตอนไม่เกิด error
อ่านตัวเลขจากไฟล์มาได้ line = "85" แล้วทำ line + 5 จะเกิดอะไรขึ้น?
ข้อมูลจากไฟล์เป็น str เสมอ — ต่อ str กับ int ไม่ได้ ต้อง int(line) + 5
ทำไมไม่ควรเขียน except: เปล่า ๆ?
ระบุชนิดที่คาดไว้เสมอ เพื่อให้ error ที่ไม่ได้คาดยังแสดงออกมาให้เห็น
ไฟล์ภาษาไทยควรเปิดยังไง?
ไม่ระบุ encoding แล้ว Windows จะเดาเอง ทำให้ภาษาไทยเพี้ยนหรือ error
ใน flowchart ของ try-except ทาง "จริง" ของเพชร "เกิด error ?" ตรงกับโค้ดส่วนไหน?
เกิด error → ไปทำบล็อก except · ไม่เกิด (เท็จ) → ทำต่อตามปกติ ซึ่งก็คือส่วน else

โจทย์ 4 — ท้าทาย: สมุดบันทึกที่ไม่หาย

ทำโปรแกรมจดบันทึกที่ข้อมูลอยู่ครบแม้ปิดโปรแกรมไปแล้ว — รวมทุกอย่างที่เรียนมา 13 บท:

  1. ตอนเปิดโปรแกรม อ่านบันทึกเดิมจาก notes.txt (ถ้ายังไม่มีไฟล์ให้เริ่มจากว่าง · ลองทั้งแบบไม่มีไฟล์ และแบบใช้ไฟล์ตัวอย่าง)
  2. เมนู: 1) เพิ่มบันทึก 2) ลบตามเลขลำดับ 3) ค้นหาคำ 4) แสดงทั้งหมด 5) ออก
  3. ทุกครั้งที่แก้ไข ให้บันทึกลงไฟล์ทันที
▸ ดูเฉลย 
FILENAME = "notes.txt"


def load():
    try:
        with open(FILENAME, encoding="utf-8") as f:
            return [line.strip() for line in f if line.strip()]
    except FileNotFoundError:
        return []


def save(notes):
    with open(FILENAME, "w", encoding="utf-8") as f:
        for n in notes:
            f.write(n + "\n")


def show(notes):
    if not notes:
        print("ยังไม่มีบันทึก")
        return
    for i, n in enumerate(notes, start=1):
        print(f"{i}. {n}")


notes = load()
print(f"โหลดบันทึกเดิม {len(notes)} รายการ")

while True:
    print("\n1) เพิ่ม  2) ลบ  3) ค้นหา  4) แสดง  5) ออก")
    c = input("เลือก: ").strip()

    if c == "1":
        notes.append(input("บันทึก: ").strip())
        save(notes)
        print("บันทึกแล้ว")

    elif c == "2":
        show(notes)
        try:
            i = int(input("ลบลำดับที่: "))
            removed = notes.pop(i - 1)      # ผู้ใช้นับจาก 1
            save(notes)
            print("ลบแล้ว:", removed)
        except ValueError:
            print("ใส่เป็นตัวเลข")
        except IndexError:
            print("ไม่มีลำดับนั้น")

    elif c == "3":
        word = input("ค้นหาคำ: ").lower()
        found = [n for n in notes if word in n.lower()]
        if found:
            show(found)
        else:
            print("ไม่พบ")

    elif c == "4":
        show(notes)

    elif c == "5":
        print("บันทึกไว้เรียบร้อย บ๊ายบาย")
        break

    else:
        print("เลือก 1-5 เท่านั้น")
