Matplotlib — พล็อตกราฟ

ตัวเลขเป็นแถวยาว ๆ ดูยังไงก็ไม่เห็นอะไร พอวาดเป็นกราฟปุ๊บ แนวโน้มกับค่าผิดปกติโผล่มาให้เห็นทันที

16.1 ติดตั้งและ import

pip install matplotlib
import matplotlib.pyplot as plt      # ทุกคนเขียนย่อว่า plt เหมือนกันหมด
ทำไมต้อง .pyplot — matplotlib เป็นก้อนใหญ่ ส่วนที่เราใช้วาดกราฟคือ pyplot เท่านั้น การ import แบบนี้จึงเป็นมาตรฐานที่เจอในโค้ดทุกที่

16.2 กราฟแรก

import matplotlib.pyplot as plt

days   = [1, 2, 3, 4, 5]
scores = [55, 62, 60, 78, 85]

plt.plot(days, scores)      # เตรียมกราฟไว้ในหน่วยความจำ
plt.show()                  # เปิดหน้าต่างแสดงผล

ใส่ชื่อและป้ายกำกับให้ครบ กราฟถึงจะอ่านรู้เรื่อง

plt.plot(days, scores, marker="o")     # marker = จุดตรงข้อมูลจริง

plt.title("Score by day")
plt.xlabel("Day")
plt.ylabel("Score")
plt.grid(True)                         # เส้นตารางช่วยกะค่า

plt.show()
ลืม plt.show() = ไม่มีอะไรขึ้นเลย — คำสั่ง plot แค่เตรียมกราฟไว้เฉย ๆ ต้องสั่ง show() ปิดท้ายทุกครั้ง

16.3 ส่วนประกอบของกราฟ

16.4 ชนิดกราฟที่ใช้บ่อย

คำสั่งกราฟแบบใช้ตอบคำถามว่า
plt.plot()เส้นเปลี่ยนแปลงยังไงตามเวลา
plt.bar()แท่งกลุ่มไหนมากกว่ากัน
plt.scatter()จุดกระจายสองค่านี้เกี่ยวกันไหม
plt.hist()ฮิสโทแกรมข้อมูลกองอยู่ช่วงไหน
plt.pie()วงกลมแต่ละส่วนคิดเป็นกี่เปอร์เซ็นต์
# แท่ง — เทียบกลุ่ม
subjects = ["Math", "Thai", "Sci", "Eng"]
scores   = [72, 85, 66, 90]
plt.bar(subjects, scores)
plt.show()

# จุดกระจาย — ดูความสัมพันธ์
hours = [1, 2, 3, 4, 5, 6, 7]
score = [40, 45, 55, 58, 70, 76, 88]
plt.scatter(hours, score)
plt.show()

# ฮิสโทแกรม — ดูการกระจาย (bins = จำนวนช่อง)
ages = [12, 13, 13, 14, 14, 14, 15, 15, 16, 18, 18, 19]
plt.hist(ages, bins=8)
plt.show()

# วงกลม — สัดส่วน
plt.pie([45, 30, 25], labels=["A", "B", "C"], autopct="%1.0f%%")
plt.show()
hist() ต่างจาก bar() — bar เราบอกความสูงของแต่ละแท่งเอง ส่วน hist รับข้อมูลดิบมาแล้วนับให้เองว่าแต่ละช่วงมีกี่ตัว

16.5 แต่งกราฟและบันทึกเป็นไฟล์

plt.figure(figsize=(8, 4))             # ขนาดภาพเป็นนิ้ว (กว้าง, สูง)

plt.plot(days, a, color="tomato", linestyle="--", marker="o", label="ห้อง ก")
plt.plot(days, b, color="steelblue", linewidth=2, label="ห้อง ข")

plt.legend()                           # ต้องมี label ก่อน legend ถึงจะขึ้น
plt.ylim(0, 100)                       # บังคับช่วงแกน y

plt.savefig("chart.png", dpi=150)      # บันทึกก่อน
plt.show()                             # แล้วค่อยแสดง
savefig ต้องมาก่อน show — เพราะ show() จะล้างกราฟทิ้งหลังปิดหน้าต่าง ถ้าสลับลำดับจะได้ไฟล์ภาพว่างเปล่า
พารามิเตอร์ตัวอย่างค่า
color"red", "tomato", "#4e6d86"
linestyle"-" ทึบ · "--" ประ · ":" จุด
marker"o" วงกลม · "s" สี่เหลี่ยม · "^" สามเหลี่ยม
labelชื่อที่จะไปโผล่ใน legend()
เรียก plt.show() ก่อน plt.savefig() จะได้ไฟล์ภาพแบบไหน?
บันทึกก่อนแสดงเสมอ · ลำดับที่ถูกคือ savefig() แล้วค่อย show()

16.6 ภาษาไทยในกราฟ

ค่าเริ่มต้นของ matplotlib ไม่มีฟอนต์ไทย ข้อความไทยจะกลายเป็นสี่เหลี่ยมเปล่า แก้ด้วยการตั้งฟอนต์ก่อนวาด

import matplotlib.pyplot as plt

plt.rcParams["font.family"] = "Tahoma"        # Windows มีอยู่แล้ว
plt.rcParams["axes.unicode_minus"] = False    # กันเครื่องหมายลบเพี้ยน

plt.bar(["คณิต", "ไทย", "วิทย์"], [72, 85, 66])
plt.title("คะแนนรายวิชา")
plt.show()
ฟอนต์ไทยที่มักมีติดเครื่อง — Windows ใช้ "Tahoma" · macOS ใช้ "Thonburi" · Linux ลอง "Noto Sans Thai" ตั้งครั้งเดียวบนสุดของไฟล์ ใช้ได้ทั้งโปรแกรม

16.7 หลายกราฟในภาพเดียว

fig, ax = plt.subplots(1, 2, figsize=(10, 4))   # 1 แถว 2 คอลัมน์

ax[0].plot(days, scores, marker="o")
ax[0].set_title("Line")                          # ในโหมดนี้ใช้ set_ นำหน้า

ax[1].bar(subjects, scores)
ax[1].set_title("Bar")

plt.tight_layout()                               # กันตัวหนังสือทับกัน
plt.show()
ชื่อคำสั่งเปลี่ยนไปนิดนึง — พอวาดผ่าน ax ต้องเติม set_ ข้างหน้า เช่น plt.title() กลายเป็น ax.set_title() และ plt.xlabel() กลายเป็น ax.set_xlabel()

16.8 ใช้คู่กับ OpenCV

เรื่องที่ต้องคำนวณเองในบทที่ 15 พอมี matplotlib ก็เห็นเป็นภาพได้ทันที

import cv2
import matplotlib.pyplot as plt

img  = cv2.imread("cat.jpg")
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# แสดงภาพ — ต้องแปลง BGR เป็น RGB ก่อน ไม่งั้นสีเพี้ยน
plt.subplot(1, 2, 1)
plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB))
plt.title("image")
plt.axis("off")                       # ภาพไม่ต้องมีแกน

# ฮิสโทแกรมของภาพเดียวกัน
plt.subplot(1, 2, 2)
hist = cv2.calcHist([gray], [0], None, [256], [0, 256])
plt.plot(hist)
plt.title("histogram")
plt.xlim(0, 255)

plt.tight_layout()
plt.show()
ลืม cvtColor = ภาพสีเพี้ยน — OpenCV เก็บสีเป็น BGR แต่ matplotlib อ่านเป็น RGB ฟ้าจะกลายเป็นส้ม (ดูบทที่ 14)
ส่วนภาพเทาต้องบอกด้วยว่าให้ใช้โทนเทา: plt.imshow(gray, cmap="gray")

16.9 Flowchart การสร้างกราฟ

การพล็อตกราฟมีลำดับตายตัว: ข้อมูล → สร้างผืน → วาด → แต่ง → บันทึก/แสดง ข้อควรจำคือ savefig ควรมาก่อน show เพราะพอปิดหน้าต่าง show แล้ว figure จะถูกปิดไปด้วย plt.savefig ที่เรียกทีหลังจึงมักได้ภาพเปล่า

fig, ax = plt.subplots()
ax.plot(x, y)
ax.set_title("ยอดขาย")
ax.set_xlabel("เดือน")
ax.set_ylabel("บาท")
if save:
    fig.savefig("plot.png")
plt.show()

แบบทดสอบ · เลือกตอบ

เลือกได้ข้อละครั้งเดียว คิดก่อนกด

เขียน import matplotlib.pyplot as plt แล้ว เวลาวาดกราฟเส้นต้องเรียกอะไร?
as plt คือการตั้งชื่อเล่นให้สั้นลง เวลาใช้จึงเรียกผ่าน plt
เขียน plt.plot(x, y) แล้วรันโปรแกรม แต่ไม่มีหน้าต่างกราฟขึ้นมา น่าจะลืมอะไร?
plot แค่เตรียมกราฟไว้ ต้อง show() ถึงจะเปิดหน้าต่าง
อยากเทียบคะแนนของ 4 วิชาว่าวิชาไหนสูงกว่ากัน ควรใช้กราฟแบบไหน?
แท่งเทียบกลุ่มกันง่ายที่สุด ส่วนวงกลมเหมาะกับการดูสัดส่วนของทั้งหมด
ใส่ชื่อแกนนอนของกราฟด้วยคำสั่งอะไร?
x คือแกนนอน y คือแกนตั้ง ส่วน title() คือชื่อกราฟด้านบน
ใส่ข้อความภาษาไทยในกราฟแล้วออกมาเป็นสี่เหลี่ยมเปล่า เพราะอะไร?
ตั้ง plt.rcParams["font.family"] = "Tahoma" บนสุดของไฟล์ก็แก้ได้
อยากให้ legend() แสดงชื่อเส้น ต้องทำอะไรตอนวาด?
legend() ไปเก็บค่า label ของแต่ละเส้นมาแสดง ถ้าไม่ใส่ก็ไม่มีอะไรให้แสดง
เอาภาพจาก OpenCV มา plt.imshow() ตรง ๆ แล้วสีเพี้ยน แก้ยังไง?
OpenCV เรียงสีเป็น BGR แต่ matplotlib อ่านเป็น RGB จึงต้องแปลงก่อน
ใน flowchart สร้างกราฟ (16.9) ทำไมกล่อง savefig ต้องมาก่อน plt.show()?
ปิดหน้าต่าง show แล้ว figure ถูกปิดตามไป plt.savefig ทีหลังจึงมักได้ภาพเปล่า · savefig ก่อนปลอดภัยเสมอ

โจทย์ 1 — กราฟเส้นอันแรก

สร้าง list วัน 1-7 กับอุณหภูมิ 7 ค่า แล้ววาดกราฟเส้นพร้อมจุด (marker="o") ใส่ชื่อกราฟ ชื่อแกนทั้งสอง และเปิดเส้นตาราง

▸ ดูเฉลย 
import matplotlib.pyplot as plt

days = [1, 2, 3, 4, 5, 6, 7]
temp = [31, 33, 30, 29, 34, 35, 32]

plt.plot(days, temp, marker="o")
plt.title("Temperature")
plt.xlabel("Day")
plt.ylabel("Celsius")
plt.grid(True)
plt.show()

โจทย์ 2 — กราฟแท่งคะแนนวิชา

เก็บคะแนนไว้ใน dict (วิชา → คะแนน) แล้ววาดกราฟแท่ง พร้อมพิมพ์ชื่อวิชาที่ได้คะแนนสูงสุดออกทางจอด้วย

▸ ดูเฉลย 
import matplotlib.pyplot as plt

scores = {"Math": 72, "Thai": 85, "Sci": 66, "Eng": 90}

plt.bar(list(scores.keys()), list(scores.values()))
plt.title("Score by subject")
plt.ylabel("Score")
plt.ylim(0, 100)

best = max(scores, key=scores.get)
print("สูงสุด:", best, scores[best])

plt.show()

โจทย์ 3 — สองเส้นในกราฟเดียว

วาดคะแนนของนักเรียน 2 คนใน 5 สัปดาห์ลงกราฟเดียวกัน ให้คนละสีคนละแบบเส้น ใส่ label แล้วเรียก legend() และบันทึกเป็น compare.png

▸ ดูเฉลย 
import matplotlib.pyplot as plt

weeks = [1, 2, 3, 4, 5]
fern  = [60, 65, 70, 68, 80]
nell  = [55, 70, 66, 78, 85]

plt.plot(weeks, fern, color="tomato",    marker="o", label="Fern")
plt.plot(weeks, nell, color="steelblue", marker="s", linestyle="--", label="Nell")

plt.title("Weekly score")
plt.xlabel("Week")
plt.ylabel("Score")
plt.legend()
plt.grid(True)

plt.savefig("compare.png", dpi=150)      # บันทึกก่อน show เสมอ
plt.show()

โจทย์ 4 — ฮิสโทแกรมของรูปจริง

อ่านรูปด้วย OpenCV แปลงเป็นเทา แล้วแสดง 2 ช่องในภาพเดียว — ซ้ายเป็นตัวรูป ขวาเป็นฮิสโทแกรมของรูปนั้น

▸ ดูเฉลย 
import cv2
import matplotlib.pyplot as plt

img = cv2.imread("cat.jpg")
if img is None:
    print("ไม่พบรูป")
    exit()

gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

plt.subplot(1, 2, 1)
plt.imshow(gray, cmap="gray")       # ภาพเทาต้องบอก cmap ด้วย
plt.title("image")
plt.axis("off")

plt.subplot(1, 2, 2)
hist = cv2.calcHist([gray], [0], None, [256], [0, 256])
plt.plot(hist)
plt.title("histogram")
plt.xlim(0, 255)

plt.tight_layout()
plt.show()
กราฟที่ดีไม่ได้แปลว่ากราฟสวย — แต่คือกราฟที่คนอ่านเข้าใจได้เองโดยไม่ต้องถาม ใส่ชื่อกราฟ ชื่อแกน และหน่วยให้ครบทุกครั้ง แค่นี้ก็ดีกว่ากราฟหลากสีที่ไม่มีป้ายอะไรเลย

โจทย์ 5 — วาด Flowchart

มี dict data เก็บชื่อสินค้ากับยอดขายรายเดือน วาดทุกสินค้าเป็นเส้นในกราฟเดียวกันพร้อม legend วาด flowchart ก่อนเขียนโค้ด

▸ ดูเฉลย 
fig, ax = plt.subplots()
for name, ys in data.items():
    ax.plot(x, ys, label=name)
ax.legend()
plt.show()
