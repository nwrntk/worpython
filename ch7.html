7.1 รูปแบบพื้นฐาน

อ่านว่า "ตราบใดที่เงื่อนไขยังเป็นจริง ให้ทำสิ่งนี้ซ้ำไปเรื่อย ๆ"

i = 1
while i <= 5:
    print(i)
    i += 1        # ห้ามลืม! ไม่งั้นวนไม่จบ

# 1 2 3 4 5

while loop ที่ถูกต้องต้องมีครบ 3 อย่างเสมอ

ส่วนทำอะไรในตัวอย่าง
1. เริ่มต้นตั้งค่าตัวแปรก่อนเข้า loopi = 1
2. เงื่อนไขเช็กทุกรอบว่าจะไปต่อไหมi <= 5
3. เปลี่ยนค่าทำให้เงื่อนไขเป็นเท็จได้สักวันi += 1
Infinite loop: ถ้าลืมข้อ 3 โปรแกรมจะวนไม่มีวันจบ
i = 1
while i <= 5:
    print(i) ← i เป็น 1 ตลอด เงื่อนไขจริงตลอด
ถ้าเผลอรันไปแล้ว กด Ctrl+C เพื่อหยุด

7.2 for หรือ while เลือกยังไง

forwhile
ใช้เมื่อรู้จำนวนรอบ / มีข้อมูลชุดหนึ่งให้ไล่ไม่รู้จำนวนรอบ รู้แค่เงื่อนไขหยุด
ตัวอย่างพิมพ์สมาชิกทุกตัวใน list, สูตรคูณ 12 บรรทัดถามรหัสผ่านจนกว่าจะถูก, เล่นเกมจนกว่าจะแพ้

ทั้งคู่เขียนแทนกันได้ แต่เลือกให้ตรงงานจะอ่านง่ายกว่า

# งานเดียวกัน เขียนได้ทั้งสองแบบ
for i in range(5):
    print(i)

i = 0

    print(i)
    i += 1
# งานนี้ for เหมาะกว่า เพราะรู้อยู่แล้วว่า 5 รอบ

7.3 งานที่ while ทำได้ดีกว่า

วนจนกว่าผู้ใช้จะกรอกถูก — ไม่มีทางรู้ว่าเขาจะกรอกผิดกี่ครั้ง

password = ""
while password != "python123":
    password = input("รหัสผ่าน: ")
    if password != "python123":
        print("ผิด ลองใหม่")
print("เข้าสู่ระบบสำเร็จ")

รับค่าไปเรื่อย ๆ จนเจอค่าที่สั่งหยุด (เรียกว่า sentinel)

total = 0
n = int(input("ใส่ตัวเลข (0 เพื่อจบ): "))

while n != 0:
    total += n
    n = int(input("ใส่ตัวเลข (0 เพื่อจบ): "))

print("ผลรวม:", total)

คำนวณจนกว่าจะถึงเป้า

# เงินฝาก 1000 ดอกเบี้ย 5% ต่อปี กี่ปีถึงจะได้ 2000?
money = 1000
year = 0

while money < 2000:
    money = money * 1.05
    year += 1

print(f"ใช้เวลา {year} ปี ได้เงิน {money:.2f} บาท")
# ใช้เวลา 15 ปี ได้เงิน 2078.93 บาท

7.4 break และ continue ใน while

# รูปแบบยอดนิยม: while True + break
while True:
    n = input("ใส่ตัวเลข (q เพื่อออก): ")
    if n == "q":
        break                # ออกจาก loop
    print("กำลังสอง:", int(n) ** 2)

print("จบโปรแกรม")

while True คือ loop ที่จริงตลอดกาล ต้องมี break เสมอไม่งั้นไม่มีวันจบ — แต่ในงานรับ input มันอ่านง่ายกว่าเงื่อนไขซับซ้อน ๆ

# continue: ข้ามรอบนี้ไป
i = 0
while i < 10:
    i += 1
    if i % 2 == 0:
        continue          # เลขคู่ไม่ต้องพิมพ์
    print(i, end=" ")
# 1 3 5 7 9
ระวัง continue ใน while: ถ้าเขียน i += 1 ไว้หลัง continue ตัวนับจะไม่ถูกเพิ่มในรอบที่ข้าม → infinite loop ทันที ในตัวอย่างข้างบนจึงเพิ่ม i ไว้บรรทัดแรกสุดของ loop

7.5 ตรวจ input ให้ไม่พัง

ถ้าผู้ใช้พิมพ์ตัวอักษรใส่ int(input()) โปรแกรมจะพังทันที ป้องกันด้วย while

while True:
    s = input("อายุ: ")
    if s.isdigit():          # เช็กว่าเป็นตัวเลขล้วนไหม
        age = int(s)
        break
    print("กรุณาใส่ตัวเลขเท่านั้น")

print("อายุของคุณคือ", age)

7.6 while กับ list

ใช้ตอนที่ต้องทยอย "ดึงของออก" จนหมด — for ทำแบบนี้ไม่ได้ดีนัก เพราะการแก้ list ระหว่างวน for จะทำให้ผลเพี้ยน

queue = ["ฟ้า", "นล", "อาร์ม", "มิน"]

while queue:                 # list ว่าง = False, มีของ = True
    person = queue.pop(0)    # ดึงคนหน้าสุดออก
    print(f"เรียก {person} (เหลือคิว {len(queue)} คน)")

print("หมดคิวแล้ว")
ค่าที่ถือว่าเป็นเท็จใน Python: 0, "", [], (), None, False
ดังนั้น while queue: จึงแปลว่า "ตราบใดที่ยังมีของใน queue"

7.7 while...else

ท่าที่พบไม่บ่อยแต่มีจริง — else จะรันเมื่อ loop จบด้วยเงื่อนไขเป็นเท็จ แต่จะไม่รันถ้าออกด้วย break

target = 7
nums = [3, 5, 9, 12]
i = 0

while i < len(nums):
    if nums[i] == target:
        print("เจอที่ index", i)
        break
    i += 1
else:
    print("ไม่พบเลข", target)

# ไม่พบเลข 7

แบบทดสอบ · เลือกตอบ

เลือกได้ข้อละครั้งเดียว คิดก่อนกด

โค้ดนี้เกิดอะไรขึ้น?
i = 0
while i < 3:
    print("hi")
ไม่มีบรรทัดเพิ่มค่า i เงื่อนไข 0 < 3 จึงจริงตลอด → ต้องมี i += 1
โค้ดนี้พิมพ์อะไร?
i = 5
while i > 0:
    print(i, end=" ")
    i -= 2
5 → 3 → 1 → -1 (ตอน i = -1 เงื่อนไขเป็นเท็จ จึงไม่พิมพ์)
งานใดควรใช้ while มากกว่า for?
ไม่รู้ว่าผู้ใช้จะกรอกผิดกี่ครั้ง → รู้แค่เงื่อนไขหยุด → while ส่วนอีก 3 ข้อรู้จำนวนรอบชัดเจน → for
while True: ต้องมีอะไรเสมอ?
เงื่อนไขจริงตลอด ต้องมีทางออกจาก loop ไม่งั้นวนไม่จบ
โค้ดนี้พิมพ์เลขอะไรบ้าง?
i = 0
while i < 5:
    i += 1
    if i == 3:
        continue
    print(i, end=" ")
รอบที่ i เป็น 3 โดน continue ข้าม print ไป แต่เพราะ i += 1 อยู่ก่อน continue จึงไม่ติด infinite loop
q = [1, 2, 3] แล้ว while q: หมายความว่าอะไร?
list ว่างถือเป็นเท็จ list ที่มีของถือเป็นจริง — ต้องมีคำสั่งลดของใน list ข้างใน loop ด้วย ไม่งั้นวนไม่จบ
ต่างกันอย่างไรระหว่าง break กับ continue?
ทั้งสองคำสั่งใช้ได้ทั้ง for และ while — ต่างกันที่ break หยุดทั้ง loop ส่วน continue แค่ข้ามรอบปัจจุบัน

โจทย์ 1 — นับถอยหลัง

ใช้ while loop พิมพ์นับถอยหลังจาก 10 ถึง 1 แล้วจบด้วยคำว่า "Go!"

▸ ดูเฉลย 
i = 10
while i >= 1:
    print(i, end=" ")
    i -= 1
print("Go!")

โจทย์ 2 — รวมเลขจนกว่าจะใส่ 0

รับตัวเลขจากผู้ใช้ไปเรื่อย ๆ เก็บลง list จนกว่าผู้ใช้จะใส่ 0 แล้วพิมพ์ list, ผลรวม, ค่าเฉลี่ย และค่ามากสุด
(ถ้าผู้ใช้ใส่ 0 ทันทีตั้งแต่แรก ให้พิมพ์ว่า "ไม่มีข้อมูล")

▸ ดูเฉลย 
nums = []

while True:
    n = int(input("ใส่ตัวเลข (0 เพื่อจบ): "))
    if n == 0:
        break
    nums.append(n)

if not nums:                     # list ว่าง
    print("ไม่มีข้อมูล")
else:
    print("ข้อมูล:", nums)
    print("ผลรวม:", sum(nums))
    print("เฉลี่ย:", round(sum(nums) / len(nums), 2))
    print("มากสุด:", max(nums))

โจทย์ 3 — เกมทายเลข

โปรแกรมสุ่มเลข 1–100 (ใช้ import random กับ random.randint(1, 100)) ให้ผู้ใช้ทายไปเรื่อย ๆ ตอบว่า "สูงไป" หรือ "ต่ำไป" ทุกครั้ง จนกว่าจะทายถูก แล้วบอกว่าใช้ไปกี่ครั้ง

▸ ดูเฉลย 
import random

secret = random.randint(1, 100)
count = 0

while True:
    guess = int(input("ทายเลข 1-100: "))
    count += 1

    if guess < secret:
        print("ต่ำไป")
    elif guess > secret:
        print("สูงไป")
    else:
        print(f"ถูกต้อง! ใช้ไป {count} ครั้ง")
        break

โจทย์ 4 — เมนูโปรแกรม

ทำโปรแกรมจัดการรายชื่อ วนแสดงเมนูไปเรื่อย ๆ จนกว่าจะเลือก 4
1) เพิ่มชื่อ   2) ลบชื่อ   3) แสดงรายชื่อทั้งหมด   4) ออก
(รวม list, for และ while เข้าด้วยกัน)

▸ ดูเฉลย 
names = []

while True:
    print("\n1) เพิ่ม  2) ลบ  3) แสดง  4) ออก")
    choice = input("เลือก: ")

    if choice == "1":
        names.append(input("ชื่อ: "))
        print("เพิ่มแล้ว")

    elif choice == "2":
        target = input("ลบชื่อ: ")
        if target in names:
            names.remove(target)
            print("ลบแล้ว")
        else:
            print("ไม่พบชื่อนี้")

    elif choice == "3":
        if not names:
            print("ยังไม่มีรายชื่อ")
        for i, n in enumerate(names, start=1):
            print(f"{i}. {n}")

    elif choice == "4":
        print("บ๊ายบาย")
        break

    else:
        print("เลือก 1-4 เท่านั้น")

โจทย์ 5 — ท้าทาย

รับเลขจำนวนเต็มบวก 1 ตัว แล้วหาผลรวมของเลขแต่ละหลัก ด้วย while loop โดยห้ามแปลงเป็น string
เช่น 4821 → 4+8+2+1 = 15
ใบ้: n % 10 ได้หลักสุดท้าย, n // 10 ตัดหลักสุดท้ายทิ้ง

▸ ดูเฉลย 
n = int(input("ใส่จำนวนเต็มบวก: "))

total = 0
temp = n            # เก็บค่าเดิมไว้พิมพ์ตอนท้าย

while temp > 0:
    total += temp % 10     # เอาหลักสุดท้าย
    temp = temp // 10      # ตัดหลักสุดท้ายทิ้ง

print(f"ผลรวมหลักของ {n} คือ {total}")

# 4821 → 1 → 2 → 8 → 4 → รวม 15
ผ่านครึ่งทางแล้ว — ถ้าทำโจทย์ทั้งหมดได้เอง แปลว่าใช้ list, tuple, for และ while ได้จริงแล้ว บทถัดไปคือ dictionary ซึ่งเก็บข้อมูลเป็นคู่ key → value และค้นด้วยชื่อได้เลย
